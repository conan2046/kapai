local Protocol = require("Protocol")
local itemCatalog = require("Data.ItemCatalog")
local itemsById = {}
for _, value in ipairs(itemCatalog) do itemsById[value.id] = value end

-- The shipped item catalog starts the equipment-fragment sequence at 4605,
-- while /320 still awards the four preceding 素心 fragments.  Their identity
-- is fixed by equip_dat.lua (1001-1004) and the contiguous 4605+ fragment
-- sequence, so keep this correction local to World instead of changing the
-- shared item catalog used by every module.
local worldMissingFragments = {
    [4601] = { name = "碎片素心刀", equipmentPicture = "petequip_2101", quality = 2 },
    [4602] = { name = "碎片素心鞋", equipmentPicture = "petequip_2102", quality = 2 },
    [4603] = { name = "碎片素心衣", equipmentPicture = "petequip_2103", quality = 2 },
    [4604] = { name = "碎片素心腕", equipmentPicture = "petequip_2104", quality = 2 },
}

local M = {
    mapType = 1,
    validationStage = nil,
    validationStageId = 0,
    validationChapterId = 0,
    validationRewardCount = 0,
    requestedStageId = 0,
    -- 龙崖改造：连战状态（op=1 下发的 fuben_AB == 2 时开启）
    chainMode = false,      -- 是否为龙崖副本模式
    -- 「自动挑战」控制失败后回本章第一关重试，以及 Boss 胜利后重跑当前章节。
    -- 普通关胜利后的下一关续战不依赖该开关；「自动挑战下一章」由结算层处理。
    chainAuto = false,
    chainIndex = 0,         -- 当前第几场（服务端回包为准）
    chainNextNodeId = 0     -- 下一关（失败时 = 本章第一关，BOSS 通关 = 0）
}

local function send(op, write)
    local message = Protocol.newMessage(Protocol.WORLD_STAGE)
    message:WriteByte(op)
    if write then write(message) end
    Bridge:Send(message)
end

local function requestWorld()
    M.requestedStageId = 0
    send(1, function(message) message:WriteByte(M.mapType) end)
end

local function requestChapter(chapterId)
    M.requestedStageId = 0
    send(2, function(message)
        message:WriteByte(M.mapType)
        message:WriteUInt(chapterId)
    end)
end

local function requestStage(stageId)
    local chapterId = Bridge:GetWorldSelectedChapterId()
    if chapterId == 0 or stageId == 0 then return end
    -- /320 op=27 is also emitted by HeroEquip probes.  Remember the World
    -- request so an unrelated same-chapter response cannot replace selection.
    M.requestedStageId = stageId
    send(27, function(message)
        message:WriteByte(M.mapType)
        message:WriteUInt(chapterId)
        message:WriteUInt(stageId)
    end)
end

local function challengeStage(stageId)
    local chapterId = Bridge:GetWorldSelectedChapterId()
    if chapterId == 0 or stageId == 0 then return end
    send(5, function(message)
        message:WriteByte(M.mapType)
        message:WriteUInt(chapterId)
        message:WriteUInt(stageId)
    end)
end

-- 龙崖模式：连战请求（/320 op=28）。语义 = 「请求本场」，只打一场，
-- 结果由服务端下发：胜利 → op=8（带本场序号 / 总场次 / 下一关），
--                   失败 → op=28 轻量回包（下一关 = 本章第一关），不弹结算。
local function challengeChainStage(stageId)
    local chapterId = Bridge:GetWorldSelectedChapterId()
    if chapterId == 0 or stageId == 0 then return end
    send(28, function(message)
        message:WriteByte(M.mapType)
        message:WriteUInt(chapterId)
        message:WriteUInt(stageId)
        -- 保留旧协议字段；连战不按次数限制，服务端忽略该字段。
        message:WriteByte(0)
    end)
end

-- 龙崖模式：连战失败回包（/320 op=28），不弹结算，仅记录下一关
local function readChainLose(message)
    local _mapId = message:ReadUInt()
    local _stageId = message:ReadUInt()
    M.chainNextNodeId = message:ReadUInt()
    M.chainIndex = message:ReadByte()
    local _chainTotal = message:ReadByte()
    if message.Remaining ~= 0 then fail("World/320 op=28 unread bytes: " .. tostring(message.Remaining)); return end
    -- op=28 的失败回包本身就是龙崖连战证据。若章节列表的模式尾字段因
    -- 旧服务端/热重载没有及时同步，不能把这场失败再落回普通副本流程。
    if M.chainIndex > 0 and not M.chainMode then
        M.chainMode = true
        Bridge:SetWorldChainMode(true)
    end
    Bridge:ApplyWorldChainLose(M.chainNextNodeId)
    -- 失败：「自动挑战」勾着 → 回本章第一关继续（服务端下发的 chainNextNodeId 即第一关）；
    --        未勾选 → 不续战，清掉本场奖励后出结算收尾，别把玩家晾在路中间。
    if M.chainAuto and M.chainNextNodeId ~= 0 then
        Bridge:ContinueWorldChain(M.chainNextNodeId)
    else
        Bridge:BeginRewardUpdate(0)
        Bridge:ShowWorldBattleResult(0)
    end
end

local function sweepStage(stageId)
    local chapterId = Bridge:GetWorldSelectedChapterId()
    if chapterId == 0 or stageId == 0 then return end
    send(6, function(message)
        message:WriteByte(M.mapType)
        message:WriteUInt(chapterId)
        message:WriteUInt(stageId)
    end)
end

local function resetStage(stageId)
    if stageId == 0 then return end
    send(7, function(message) message:WriteUInt(stageId) end)
end

local function claimBox(boxId)
    local chapterId = Bridge:GetWorldSelectedChapterId()
    if chapterId == 0 or boxId == 0 then return end
    send(4, function(message)
        message:WriteByte(M.mapType)
        message:WriteUInt(chapterId)
        message:WriteUInt(boxId)
    end)
end

local function requestAchievement()
    send(11)
end

local function claimAchievement(index)
    if index < 1 or index > 6 then return end
    send(12, function(message) message:WriteByte(index) end)
end

local function rewardDefinition(id)
    return itemsById[id] or worldMissingFragments[id] or {}
end

local function readReward(message)
    local rewardType = message:ReadWord()
    local rewardId = message:ReadUInt()
    local amount = message:ReadUInt()
    -- Currency rewards use their 600xx award type with an authoritative payload id of 0.
    -- Keep the payload id intact, but resolve display metadata through the type in that case.
    local displayId = rewardId > 0 and rewardId or rewardType
    local found = rewardDefinition(displayId)
    local displayName = found.name or ""
    if displayName == "" then displayName = "奖励 #" .. tostring(displayId) end
    return rewardType, rewardId, amount, displayName, found.pic or 0, found.quality or 0
end

local function fail(detail)
    if M.validationStage then Bridge:Fail(detail); M.validationStage = nil
    else Bridge:SetWorldError(detail) end
end

function M.onClick()
    if Bridge:HasCommandLineFlag("-projectXWorldBattleValidation") and M.validationStage == "reconnect_pending" then
        return
    end
    Bridge:ShowWorld()
    M.validationStage = Bridge:HasCommandLineFlag("-projectXWorldBattleValidation") and "await_world" or nil
    M.validationStageId = 0
    M.validationRewardCount = 0
    requestWorld()
    Bridge:SetStatus("World/320 chapter list requested.")
end

function M.requestChapter(chapterId) requestChapter(chapterId) end
function M.requestStage(stageId) requestStage(stageId) end
function M.sweepSelected() sweepStage(Bridge:GetWorldSelectedStageId()) end
function M.resetStage(stageId) resetStage(stageId) end
function M.claimBox(boxId) claimBox(boxId) end
function M.requestAchievement() requestAchievement() end
function M.claimAchievement(index) claimAchievement(index) end
function M.queryHeroEquipmentStages()
    Bridge:BeginHeroEquipmentStageChecks()
    local stages = {
        {1001, 10006},
        {1002, 10016},
        {1002, 10019},
        {1002, 10020},
    }
    for _, value in ipairs(stages) do
        send(27, function(message)
            message:WriteByte(1)
            message:WriteUInt(value[1])
            message:WriteUInt(value[2])
        end)
    end
end
function M.openPreferredStage()
    requestStage(Bridge:GetWorldPreferredStageId())
end
function M.challengeSelected()
    local stageId = Bridge:GetWorldPreferredStageId()
    if stageId == 0 then fail("World/320 has no unlocked stage to challenge."); return end
    M.validationStageId = stageId
    M.validationStage = M.validationStage and "await_battle" or nil
    if M.chainMode then
        M.chainIndex = 1
        M.chainNextNodeId = 0
        challengeChainStage(stageId)
    else
        challengeStage(stageId)
    end
end

-- 龙崖模式：由 Unity 在单场回放播完、走位到下一目标点之后调用。
-- 返回 true 表示已发起下一场；false 表示不再继续（非龙崖模式 / 已通关）。
-- 普通关胜利后的本章顺序续战不受 chainAuto 限制；失败重试由 chainAuto 控制。
-- 本章 Boss 后是否切下一章由 chainAutoNext 控制。
function M.continueChain()
    if not M.chainMode then return false end
    local nextStage = M.chainNextNodeId
    if nextStage == 0 then return false end
    challengeChainStage(nextStage)
    return true
end

-- 龙崖模式：本章 Boss 胜利后，只勾「自动挑战」时从本章第一关重新开始。
-- 不使用次数限制；旧协议字段仅保留兼容位。
function M.restartChain()
    local chapterId = Bridge:GetWorldSelectedChapterId()
    if chapterId == 0 then return false end
    send(28, function(message)
        message:WriteByte(M.mapType)
        message:WriteUInt(chapterId)
        message:WriteUInt(0)
        message:WriteByte(0)
    end)
    return true
end

function M.cancelChain()
    if not M.chainMode then return end
    M.chainNextNodeId = 0
    send(29, function(message) message:WriteByte(M.mapType) end)
end

function M.setChainAuto(value) M.chainAuto = value and true or false end
function M.setChainAutoNext(value) M.chainAutoNext = value and true or false end
function M.refreshAfterBattle()
    M.validationStage = "verify_world"
    requestWorld()
end
function M.validateAccountIsolation()
    M.validationStage = "await_isolation_world"
    requestWorld()
end

local function readWorld(message)
    local mapType = message:ReadByte()
    local success = message:ReadByte()
    if success == 0 then fail("World/320 op=1 failed: " .. (message.Remaining > 0 and message:ReadString() or "unknown")); return end
    local count = message:ReadWord()
    Bridge:BeginWorldChapterList(mapType, count)
    for _ = 1, count do
        -- Read each wire field before calling the bridge.  xLua argument evaluation
        -- may otherwise consume the stream out of protocol order.
        local chapterId = message:ReadUInt()
        local chapterName = message:ReadString()
        local openLevel = message:ReadWord()
        local maxStars = message:ReadByte()
        Bridge:AddWorldChapter(chapterId, chapterName, openLevel, maxStars)
    end
    local currentChapter = message:ReadUInt()
    local currentStage = message:ReadUInt()
    local progressCount = message:ReadWord()
    for _ = 1, progressCount do
        local chapterId = message:ReadUInt()
        local stars = message:ReadWord()
        local claimedBoxes = message:ReadByte()
        Bridge:SetWorldChapterProgress(chapterId, stars, claimedBoxes)
    end
    -- 龙崖改造：模式下发（方案甲）—— op=1 载荷末尾的 fuben_AB，2 = 龙崖副本模式
    M.chainMode = message.Remaining > 0 and message:ReadByte() == 2 or false
    if message.Remaining ~= 0 then fail("World/320 op=1 unread bytes: " .. tostring(message.Remaining)); return end
    -- 注意：必须在 EndWorldChapterList（内部 EnsureWorldPresenter）之后下发模式，
    -- 否则 WorldPresenter 尚未创建，模式会被丢掉。
    Bridge:EndWorldChapterList(currentChapter, currentStage)
    Bridge:SetWorldChainMode(M.chainMode)
    if M.validationStage == "await_isolation_world" then
        M.validationStage = "await_isolation_chapter"
        requestChapter(currentChapter)
        return
    end
    -- 龙崖改造：进副本先停在章节选择页（chapterPage），由玩家手动选章节后
    -- 再进入大底图。原「自动打开当前章节」逻辑移除。
end

local function readChapter(message)
    local mapType = message:ReadByte()
    local chapterId = message:ReadUInt()
    local success = message:ReadByte()
    if success == 0 then fail("World/320 op=2 failed: " .. (message.Remaining > 0 and message:ReadString() or "unknown")); return end
    local chapterName = message:ReadString()
    local count = message:ReadByte()
    Bridge:BeginWorldStageList(mapType, chapterId, chapterName, count)
    for _ = 1, count do
        local stageId = message:ReadUInt()
        local stageName = message:ReadString()
        local stars = message:ReadByte()
        local attempts = message:ReadByte()
        local spirit = message:ReadByte()
        local resets = message:ReadByte()
        local resetCost = message:ReadWord()
        local nextStage = message:ReadUInt()
        local boxId = message:ReadUInt()
        local boxState = boxId > 0 and message:ReadByte() or 0
        Bridge:BeginWorldStage(stageId, stageName, stars, attempts, spirit, resets, resetCost, nextStage, boxId, boxState)
        local currencyCount = message:ReadByte()
        for _ = 1, currencyCount do
            local reward = { readReward(message) }
            Bridge:AddWorldStageReward(reward[1], reward[2], reward[3], reward[4], reward[5], reward[6], true)
        end
        local itemCount = message:ReadByte()
        for _ = 1, itemCount do
            local reward = { readReward(message) }
            Bridge:AddWorldStageReward(reward[1], reward[2], reward[3], reward[4], reward[5], reward[6], false)
        end
        Bridge:EndWorldStage()
    end
    local starBoxCount = message:ReadByte()
    for _ = 1, starBoxCount do
        local requiredStars = message:ReadByte()
        local boxId = message:ReadUInt()
        local state = message:ReadByte()
        Bridge:AddWorldStarBox(requiredStars, boxId, state)
    end
    if message.Remaining ~= 0 then fail("World/320 op=2 unread bytes: " .. tostring(message.Remaining)); return end
    Bridge:EndWorldStageList()
    if M.validationStage == "await_isolation_chapter" then
        M.validationStage = "await_isolation_stage"
        M.openPreferredStage()
        return
    end
    if M.validationStage == "await_world" then
        -- Chapter navigation inside the capture coroutine produces additional
        -- op=2 replies.  Transition before starting it so only this first
        -- authoritative chapter reply owns the validation flow.
        M.validationStage = "map_capture_running"
        Bridge:CaptureWorldMapAndContinue()
    elseif M.validationStage == "verify_world" then
        M.validationStage = "verify_stage"
        requestStage(M.validationStageId)
    else
        -- A normal /320 op=2 response only opens kapaiguaiwuLayer.  The Cocos
        -- flow does not request op=27 or show guanqiaxiangxiLayer until the
        -- player really clicks a stage node.  Keeping requestedStageId at zero
        -- also prevents a late unrelated op=27 response from opening detail.
        M.requestedStageId = 0
    end
end

local function readStage(message, responseMapType)
    local mapType = responseMapType or message:ReadByte()
    local chapterId = message:ReadUInt()
    local stageId = message:ReadUInt()
    local stars = message:ReadByte()
    local foughtCount = message:ReadByte()
    local resetCount = message:ReadByte()
    if message.Remaining ~= 0 then fail("World/320 op=27 unread bytes: " .. tostring(message.Remaining)); return end
    Bridge:SetHeroEquipmentStageOpen(stageId, stars)
    -- Hero's equipment-open probes deliberately reuse /320 op=27, including
    -- nodes in the currently visible World chapter.  They are valid Hero
    -- evidence, but only the outstanding World stage request may alter World.
    if chapterId ~= Bridge:GetWorldSelectedChapterId() then return end
    if M.requestedStageId == 0 or stageId ~= M.requestedStageId then return end
    M.requestedStageId = 0
    Bridge:SetWorldStageStatus(mapType, chapterId, stageId, stars, foughtCount, resetCount)
    local isSelectedStage = chapterId == Bridge:GetWorldSelectedChapterId()
        and stageId == Bridge:GetWorldPreferredStageId()
    if M.validationStage == "map_capture_running" and isSelectedStage then
        -- /320 op=27 is also used by unrelated World consumers (for example
        -- HeroEquip's stage probes).  Only the stage selected by this World
        -- run may enter the detail/battle validation path.
        M.validationStage = "detail_capture_running"
        Bridge:CaptureWorldDetailAndChallenge()
    elseif M.validationStage == "verify_stage" then
        M.validationStage = "reconnect_pending"
        Bridge:CompleteWorldBattleValidation(M.validationStageId, M.validationRewardCount)
    elseif M.validationStage == "await_isolation_stage" then
        M.validationStage = nil
        Bridge:CompleteWorldAccountIsolationValidation(chapterId, stageId, stars)
    end
end

local function readBattleResult(message)
    local foughtCount = message:ReadByte()
    local foughtStageId = message:ReadUInt()
    local unlockedChapterId = message:ReadUInt()
    local unlockedStageId = message:ReadUInt()
    local unlockedBoxId = message:ReadUInt()
    local unlockedStarBoxId = message:ReadUInt()
    local stars = message:ReadByte()
    -- 龙崖改造：连战序号 / 总场次 / 下一关（非连战时为 0 / 0 / 0）
    local chainIndex = message:ReadByte()
    local _chainTotal = message:ReadByte()
    M.chainIndex = chainIndex
    M.chainNextNodeId = message:ReadUInt()
    -- 连战 op=28 的胜利回包带本场序号；它比 op=1 的模式尾字段更接近
    -- 当前请求事实，避免模式状态短暂丢失时首个普通怪被误当成最终结算。
    if chainIndex > 0 and not M.chainMode then
        M.chainMode = true
        Bridge:SetWorldChainMode(true)
    end
    local rewardCount = message:ReadByte()
    Bridge:BeginRewardUpdate(rewardCount)
    local visibleRewardCount = 0
    for _ = 1, rewardCount do
        local reward = { readReward(message) }
        if reward[3] > 0 then visibleRewardCount = visibleRewardCount + 1 end
        Bridge:AddRewardRecord(reward[1], reward[2], reward[3], reward[4], reward[5], reward[6])
    end
    if message.Remaining ~= 0 then fail("World/320 op=8 unread bytes: " .. tostring(message.Remaining)); return end
    Bridge:ApplyWorldBattleResult(foughtCount, foughtStageId, unlockedChapterId, unlockedStageId,
        unlockedBoxId, unlockedStarBoxId, stars, chainIndex, _chainTotal, M.chainNextNodeId)
    M.validationRewardCount = visibleRewardCount
    -- op=8 只表示本场胜利：只要不是 Boss（nextNodeId ~= 0），无论是否勾选
    -- 「自动挑战」都要按顺序挑战本章下一关。CheckBox_1 只控制失败后的重试，
    -- 以及 Boss 胜利后的当前章节循环；CheckBox_2 由结算层处理切章。
    if (M.chainMode or chainIndex > 0) and M.chainNextNodeId ~= 0 then
        Bridge:ContinueWorldChain(M.chainNextNodeId)
    else
        Bridge:ShowWorldBattleResult(stars)
    end
    if M.validationStage == "await_battle" then
        M.validationStage = "await_result_capture"
        Bridge:CaptureWorldBattleAndRefresh(visibleRewardCount)
    end
end

local function readBoxAward(message)
    message:ReadByte()
    local chapterId = message:ReadUInt()
    local boxId = message:ReadUInt()
    local success = message:ReadByte()
    if success == 0 then
        fail("World/320 op=4 failed: " .. (message.Remaining > 0 and message:ReadString() or "unknown"))
        return
    end
    local rewardCount = message:ReadByte()
    Bridge:BeginRewardUpdate(rewardCount)
    for _ = 1, rewardCount do Bridge:AddRewardRecord(readReward(message)) end
    if message.Remaining ~= 0 then fail("World/320 op=4 unread bytes: " .. tostring(message.Remaining)); return end
    Bridge:ApplyWorldBoxClaim(chapterId, boxId)
    Bridge:EndRewardUpdate("宝箱奖励")
end

local function readSweep(message)
    message:ReadByte()
    message:ReadUInt()
    local stageId = message:ReadUInt()
    local success = message:ReadByte()
    if success == 0 then
        fail("World/320 op=6 failed: " .. (message.Remaining > 0 and message:ReadString() or "unknown"))
        return
    end
    local sweepCount = message:ReadByte()
    local rewards = {}
    Bridge:BeginWorldSweepRewards(sweepCount)
    for sweepIndex = 1, sweepCount do
        message:ReadByte()
        local currencyCount = message:ReadByte()
        for _ = 1, currencyCount do
            local reward = { readReward(message) }
            table.insert(rewards, reward)
            Bridge:AddWorldSweepReward(sweepIndex, table.unpack(reward))
        end
        local itemCount = message:ReadByte()
        for _ = 1, itemCount do
            local reward = { readReward(message) }
            table.insert(rewards, reward)
            Bridge:AddWorldSweepReward(sweepIndex, table.unpack(reward))
        end
    end
    if message.Remaining ~= 0 then fail("World/320 op=6 unread bytes: " .. tostring(message.Remaining)); return end
    Bridge:BeginRewardUpdate(#rewards)
    for _, reward in ipairs(rewards) do Bridge:AddRewardRecord(table.unpack(reward)) end
    Bridge:ApplyWorldSweep(stageId, sweepCount)
    Bridge:ShowWorldSweepResult(sweepCount)
end

local function readReset(message)
    local stageId = message:ReadUInt()
    local success = message:ReadByte()
    if success == 0 then
        fail("World/320 op=7 failed: " .. (message.Remaining > 0 and message:ReadString() or "unknown"))
        return
    end
    local usedResets = message:ReadByte()
    local cost = message:ReadWord()
    if message.Remaining ~= 0 then fail("World/320 op=7 unread bytes: " .. tostring(message.Remaining)); return end
    Bridge:ApplyWorldReset(stageId, usedResets, cost)
end

function M.onPacket(command, message)
    if command ~= Protocol.WORLD_STAGE then return false end
    local op = message:ReadByte()
    -- Local initialization also emits the one-byte legacy achievement marker
    -- (`/320 op=2`).  It has no chapter payload and must not be confused with
    -- a requested chapter response.
    -- A real op=2 result needs mapType(1), chapterId(4), success(1), an
    -- encoded name and the star-box count, i.e. at least 10 bytes after op.
    -- Shorter packets are the initialization marker, not a chapter reply.
    if op == 2 and message.Remaining < 10 then return true end
    -- The legacy server consumes op=27 before echoing the request fields, so
    -- node-query responses begin with mapType and contain exactly 11 bytes more.
    if op == 1 and message.Remaining == 11 then readStage(message, op)
    elseif op == 1 then readWorld(message)
    elseif op == 2 then readChapter(message)
    elseif op == 5 then
        message:ReadByte(); message:ReadUInt(); message:ReadUInt()
        -- `op=5` only acknowledges the fast-fight start. The server emits its
        -- authoritative stars, attempt count and rewards separately in op=8;
        -- these trailing legacy start bytes have no World-store meaning.
        while message.Remaining > 0 do message:ReadByte() end
    elseif op == 4 then readBoxAward(message)
    elseif op == 6 then readSweep(message)
    elseif op == 7 then readReset(message)
    elseif op == 8 then readBattleResult(message)
    elseif op == 28 then readChainLose(message)
    elseif op == 11 then
        local success = message:ReadByte()
        if success == 0 then fail("World/320 op=11 failed: " .. (message.Remaining > 0 and message:ReadString() or "unknown")); return true end
        local orderType = message:ReadByte()
        local bitMap = message:ReadByte()
        if message.Remaining ~= 0 then fail("World/320 op=11 unread bytes: " .. tostring(message.Remaining)); return true end
        Bridge:SetWorldAchievementState(orderType, bitMap)
    elseif op == 12 then
        local index = message:ReadByte()
        local success = message:ReadByte()
        if success == 0 then fail("World/320 op=12 failed: " .. (message.Remaining > 0 and message:ReadString() or "unknown")); return true end
        local orderType = message:ReadByte()
        local bitMap = message:ReadByte()
        local rewardType = message:ReadWord()
        local amount = message:ReadUInt()
        if message.Remaining ~= 0 then fail("World/320 op=12 unread bytes: " .. tostring(message.Remaining)); return true end
        Bridge:ApplyWorldAchievementClaim(index, orderType, bitMap, rewardType, amount)
    elseif op == 27 then readStage(message)
    else return true end
    return true
end

return M
