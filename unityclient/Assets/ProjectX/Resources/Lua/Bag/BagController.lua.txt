local Protocol = require("Protocol")
local catalog = require("Data.ItemCatalog")
local byId = {}
for _, value in ipairs(catalog) do byId[value.id] = value end

local USE_VALIDATION_ITEM_ID = 3201
local M = {
    requested = false,
    awaitingSort = false,
    validationStage = nil,
    validationSlot = nil,
    validationQuantity = nil,
    g4Stage = nil,
    drawClosureRequested = false,
    drawHeaderRequested = false
}

local function clean(value)
    value = tostring(value or "")
    value = string.gsub(value, "[\t\r\n]", " ")
    return value
end

local function encodeSources(cfg)
    local values = {}
    for _, source in ipairs(cfg.item_source or {}) do
        if source[1] ~= nil then table.insert(values, tostring(source[1])) end
    end
    return table.concat(values, ",")
end

local function encodeChoices(cfg)
    local rows = {}
    for _, reward in ipairs(cfg.sub_value or {}) do
        local id = tonumber(reward[1]) or 0
        local rewardCfg = byId[id] or {}
        local quantity = tonumber(reward[#reward]) or 1
        table.insert(rows, table.concat({
            tostring(id),
            clean(rewardCfg.name),
            clean(rewardCfg.des),
            tostring(rewardCfg.pic or 0),
            tostring(rewardCfg.quality or 0),
            tostring(rewardCfg.type or 0),
            tostring(quantity),
            clean(rewardCfg.item_from),
            encodeSources(rewardCfg)
        }, "\t"))
    end
    return table.concat(rows, "\n")
end

local function addOrUpdate(slot, id, num, replace)
    local cfg = byId[id] or {}
    if replace then
        Bridge:AddBagItem(slot, id, num, cfg.name or "", cfg.des or "", cfg.pic or 0,
            cfg.quality or 0, cfg.use_type or 0, cfg.use_jump or 0, cfg.sort_priority or 999999,
            cfg.type or 0, cfg.item_from or "", encodeChoices(cfg), encodeSources(cfg))
    else
        Bridge:UpsertBagItem(slot, id, num, cfg.name or "", cfg.des or "", cfg.pic or 0,
            cfg.quality or 0, cfg.use_type or 0, cfg.use_jump or 0, cfg.sort_priority or 999999,
            cfg.type or 0, cfg.item_from or "", encodeChoices(cfg), encodeSources(cfg))
    end
end

local function readFullPackage(message, hasTailFlags, headerOnly)
    local count = message:ReadWord()
    if headerOnly then Bridge:BeginBagHeaderUpdate(count) else Bridge:BeginBagUpdate(count) end
    for _ = 1, count do
        local slot = message:ReadWord() + 1
        local id = message:ReadWord()
        local num = message:ReadWord()
        addOrUpdate(slot, id, num, true)
    end
    if hasTailFlags and message.Remaining > 0 then message:ReadByte() end
    if hasTailFlags and message.Remaining > 0 then message:ReadByte() end
    if headerOnly then Bridge:EndBagHeaderUpdate() else Bridge:EndBagUpdate() end
    return count
end

function M.onClick()
    M.requested = true
    Bridge:Send(Protocol.newMessage(Protocol.ROLE_PACKAGE))
end

function M.requestDrawClosureSnapshot()
    M.drawClosureRequested = true
    Bridge:Send(Protocol.newMessage(Protocol.ROLE_PACKAGE))
end

-- HappyDrawUI reads the live package for its three recruitment-ticket counts.
-- Unity has no legacy LRoleDataMgr cache, so refresh the same server-owned
-- package before rendering Draw instead of inventing local counts.
function M.requestDrawHeaderRefresh()
    M.drawHeaderRequested = true
    Bridge:Send(Protocol.newMessage(Protocol.ROLE_PACKAGE))
end

function M.onDisconnected()
    M.requested = false
    M.awaitingSort = false
    if M.g4Stage ~= "await_reconnect" then M.g4Stage = nil end
end

function M.onUse(slot, quantity, target)
    local currentQuantity = Bridge:GetBagQuantity(slot)
    if currentQuantity <= 0 then
        Bridge:SetStatus("Selected bag slot is empty.")
        return
    end
    local message = Protocol.newMessage(Protocol.PACKAGE_UPDATE)
    message:WriteByte(1)
    message:WriteWord(slot - 1)
    message:WriteByte(math.max(1, math.min(tonumber(quantity) or 1, currentQuantity, 255)))
    message:WriteByte(math.max(0, math.min(tonumber(target) or 0, 255)))
    Bridge:Send(message)
    Bridge:SetStatus("Use-item request sent for slot " .. tostring(slot)
        .. ", quantity=" .. tostring(quantity) .. ", target=" .. tostring(target) .. ".")
end

local function beginUseValidation()
    local message = Protocol.newMessage(Protocol.INTERACT)
    message:WriteByte(50)
    message:WriteUInt(USE_VALIDATION_ITEM_ID)
    message:WriteWord(1)
    M.validationStage = "await_add"
    Bridge:Send(message)
    Bridge:SetStatus("Use-item validation: waiting for local item injection update.")
end

local function validateIncrementalUpdate(slot, itemId, quantity)
    if M.validationStage == "await_add" and itemId == USE_VALIDATION_ITEM_ID then
        if quantity < 1 then
            Bridge:Fail("Use-item validation received an empty injected stack.")
            M.validationStage = nil
            return
        end
        M.validationSlot = slot
        M.validationQuantity = quantity
        M.validationStage = "await_consume"
        M.onUse(slot, quantity)
        return
    end

    if M.validationStage ~= "await_consume" or slot ~= M.validationSlot then return end
    local remaining = itemId == USE_VALIDATION_ITEM_ID and quantity or 0
    if remaining ~= M.validationQuantity - 1 then
        Bridge:Fail("Use-item validation quantity mismatch: expected "
            .. tostring(M.validationQuantity - 1) .. ", got " .. tostring(remaining) .. ".")
        M.validationStage = nil
        return
    end
    M.validationStage = nil
    Bridge:Complete("COMPLETE: package/8 -> update/15 sort -> local add 3201 -> use-item request -> "
        .. "fixed reward 60015 x10 -> update/15 consume")
end

function M.onPacket(command, message)
    if command == Protocol.ROLE_PACKAGE and (M.requested or M.drawClosureRequested or M.drawHeaderRequested) then
        -- A main-screen click can still carry an obsolete package request from
        -- the imported navigation layer. Once Draw is already open, its /8
        -- response is strictly header data and must never replace the Draw UI.
        local headerOnly = M.drawHeaderRequested or Bridge:IsDrawActive()
        local serverCount = readFullPackage(message, true, headerOnly)
        local renderedCount = Bridge:GetBagCount()
        if M.drawHeaderRequested then
            M.drawHeaderRequested = false
            return true
        end
        if M.drawClosureRequested then
            M.drawClosureRequested = false
            Bridge:CompleteDrawClosureBagRefresh(serverCount)
            return true
        end
        M.requested = false
        if M.g4Stage == "await_reopen_full" then
            M.g4Stage = "await_reconnect"
            Bridge:BeginBagDisconnectValidation()
            return true
        end
        if M.g4Stage == "await_reconnect" then
            M.g4Stage = nil
            Bridge:CompleteBagG4Validation()
            return true
        end
        if Bridge:IsAutomation() then
            local sortMessage = Protocol.newMessage(Protocol.PACKAGE_UPDATE)
            sortMessage:WriteByte(6)
            M.awaitingSort = true
            Bridge:Send(sortMessage)
            Bridge:SetStatus("Package/8 rendered; waiting for safe package-update/15 sort refresh.")
            return true
        end
        Bridge:Complete("COMPLETE: ProjectXApp -> parsed package/8 -> rendered real bag UI ("
            .. tostring(renderedCount) .. " stacks, server=" .. tostring(serverCount) .. ")")
        return true
    end

    if command ~= Protocol.PACKAGE_UPDATE then return false end
    local op = message:ReadByte()
    if op == 6 then
        local success = message:ReadByte()
        if success == 1 then
            readFullPackage(message, false)
            local renderedCount = Bridge:GetBagCount()
            if M.g4Stage == "await_final_sort" then
                M.awaitingSort = false
                M.g4Stage = "await_reopen_full"
                Bridge:BeginBagReloadValidation()
                return true
            end
            if M.awaitingSort then
                M.awaitingSort = false
                if Bridge:HasCommandLineFlag("-projectXUseItemValidation") then
                    beginUseValidation()
                elseif Bridge:HasCommandLineFlag("-projectXBagG3Validation") then
                    Bridge:CompleteBagG3Validation()
                elseif Bridge:HasCommandLineFlag("-projectXBagG4Validation") then
                    M.g4Stage = "await_batch"
                    Bridge:BeginBagG4Validation()
                else
                    Bridge:Complete("COMPLETE: package/8 + package-update/15 sort -> rendered bag UI ("
                        .. tostring(renderedCount) .. " stacks)")
                end
            end
        elseif M.awaitingSort then
            M.awaitingSort = false
            Bridge:Fail("Package-update/15 sort refresh failed.")
        end
        return true
    end
    if op == 0 or op == 7 then
        message:ReadWord()
        message:ReadWord()
        local success = message:ReadByte()
        if success == 0 and message.Remaining > 0 then Bridge:SetStatus(message:ReadString()) end
        return true
    end

    local slot = message:ReadWord() + 1
    local previousItemId = Bridge:GetBagItemId(slot)
    local previousQuantity = Bridge:GetBagQuantity(slot)
    message:ReadWord()
    message:ReadWord()
    local itemId = message:ReadWord()
    if itemId == 0 then
        Bridge:RemoveBagItem(slot)
    else
        local quantity = message:ReadWord()
        if message.Remaining >= 2 then message:ReadWord() end
        addOrUpdate(slot, itemId, quantity, false)
        local added = previousItemId == itemId and (quantity - previousQuantity) or quantity
        if added > 0 then
            local cfg = byId[itemId] or {}
            Bridge:QueueBagUseReward(itemId, added, cfg.name or "", cfg.pic or 0,
                cfg.quality or 0, cfg.type or 0)
        end
    end
    validateIncrementalUpdate(slot, itemId, itemId == 0 and 0 or Bridge:GetBagQuantity(slot))
    if M.g4Stage == "await_batch" and (itemId == 500 or itemId == 0) then
        M.g4Stage = "await_gift"
        Bridge:ContinueBagG4AfterBatchUse()
    elseif M.g4Stage == "await_gift" and (itemId == 1111 or itemId == 0) then
        local inject = Protocol.newMessage(Protocol.INTERACT)
        inject:WriteByte(50)
        inject:WriteUInt(USE_VALIDATION_ITEM_ID)
        inject:WriteWord(1)
        M.g4Stage = "await_direct_add"
        Bridge:Send(inject)
    elseif M.g4Stage == "await_direct_add" and itemId == USE_VALIDATION_ITEM_ID then
        M.validationSlot = slot
        M.g4Stage = "await_direct_consume"
        Bridge:RunBagG4DirectUse()
    elseif M.g4Stage == "await_direct_consume" and slot == M.validationSlot then
        M.onUse(slot, 1, 0)
        local sortMessage = Protocol.newMessage(Protocol.PACKAGE_UPDATE)
        sortMessage:WriteByte(6)
        M.g4Stage = "await_final_sort"
        M.awaitingSort = true
        Bridge:Send(sortMessage)
    end
    return true
end

return M
