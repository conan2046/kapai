local M = {}
function M.show(autoInvoke) Bridge:ShowLoginUi(); Bridge:BindLoginClick(autoInvoke) end
function M.showRoleCreate(autoInvoke) Bridge:BindRoleCreateClick(autoInvoke) end
function M.showMain(autoInvokeBag)
    Bridge:ShowMainUi()
    local taskValidation = Bridge:HasCommandLineFlag("-projectXTaskValidation")
        or Bridge:HasCommandLineFlag("-projectXTaskMutationValidation")
    local playerValidation = Bridge:HasCommandLineFlag("-projectXPlayerHudValidation")
    local heroValidation = Bridge:HasCommandLineFlag("-projectXHeroValidation")
        or Bridge:HasCommandLineFlag("-projectXHeroBagValidation")
        or Bridge:HasCommandLineFlag("-projectXHeroEquipValidation")
        or Bridge:HasCommandLineFlag("-projectXHeroEquipMutationValidation")
    local mailValidation = Bridge:HasCommandLineFlag("-projectXMailValidation")
    local shopValidation = Bridge:HasCommandLineFlag("-projectXShopValidation")
    local friendValidation = Bridge:HasCommandLineFlag("-projectXFriendValidation")
    local chatValidation = Bridge:HasCommandLineFlag("-projectXChatValidation")
    local teamValidation = Bridge:HasCommandLineFlag("-projectXTeamValidation")
    local guildValidation = Bridge:HasCommandLineFlag("-projectXGuildValidation")
    local worldValidation = Bridge:HasCommandLineFlag("-projectXWorldBattleValidation")
    local welfareValidation = Bridge:HasCommandLineFlag("-projectXWelfareValidation")
    local activityValidation = Bridge:HasCommandLineFlag("-projectXActivityValidation")
    local drawValidation = Bridge:HasCommandLineFlag("-projectXDrawValidation")
    local gameplayValidation = Bridge:HasCommandLineFlag("-projectXGameplayValidation")
        or Bridge:HasCommandLineFlag("-projectXYouLiValidation")
        or Bridge:HasCommandLineFlag("-projectXFengShenStoryValidation")
        or Bridge:HasCommandLineFlag("-projectXArenaValidation")
        or Bridge:HasCommandLineFlag("-projectXKunLunValidation")
        or Bridge:HasCommandLineFlag("-projectXBloodFightValidation")
        or Bridge:HasCommandLineFlag("-projectXXunBaoValidation")
        or Bridge:HasCommandLineFlag("-projectXSevenDayValidation")
        or Bridge:HasCommandLineFlag("-projectXGameplayShopsValidation")
        or Bridge:HasCommandLineFlag("-projectXStaminaClaimValidation")
        or Bridge:HasCommandLineFlag("-projectXResourceRecoveryValidation")
        or Bridge:HasCommandLineFlag("-projectXFundsValidation")
    local loginValidation = Bridge:HasCommandLineFlag("-projectXLoginValidation")
    Bridge:BindBagClick(autoInvokeBag and not taskValidation and not playerValidation and not heroValidation and not mailValidation and not shopValidation and not friendValidation and not chatValidation and not teamValidation and not guildValidation and not worldValidation and not welfareValidation and not activityValidation and not drawValidation and not gameplayValidation and not loginValidation)
    Bridge:BindSettingsClick()
    Bridge:BindTaskClick(autoInvokeBag and taskValidation)
    Bridge:BindHeroClick(autoInvokeBag and heroValidation)
    Bridge:BindMailClick(autoInvokeBag and mailValidation)
    Bridge:BindShopClick(autoInvokeBag and shopValidation)
    Bridge:BindFriendClick(autoInvokeBag and friendValidation)
    Bridge:BindChatClick(autoInvokeBag and chatValidation)
    Bridge:BindTeamClick(autoInvokeBag and teamValidation)
    Bridge:BindGuildClick(autoInvokeBag and guildValidation)
    Bridge:BindWorldClick(autoInvokeBag and worldValidation)
    Bridge:BindWelfareClick(autoInvokeBag and welfareValidation)
    Bridge:BindActivityClick(autoInvokeBag and activityValidation)
    Bridge:BindDrawClick(autoInvokeBag and drawValidation)
    Bridge:BindGameplayClick(autoInvokeBag and gameplayValidation)
end
return M
