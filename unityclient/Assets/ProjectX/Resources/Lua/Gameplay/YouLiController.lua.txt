local Protocol = require("Protocol")
local HeroController = require("Hero.HeroController")
local M = { validation = false }

local function requestInfo()
    local message = Protocol.newMessage(Protocol.YOU_LI); message:WriteByte(1); Bridge:Send(message)
end

local function skipRewardBatch(message)
    local count = message:ReadByte()
    for _ = 1, count do message:ReadWord(); message:ReadUInt(); message:ReadUInt() end
end

function M.onClick()
    M.validation = Bridge:HasCommandLineFlag("-projectXYouLiValidation")
    Bridge:ShowYouLi()
    -- YouLi can be entered directly from the gameplay hall before the hero
    -- screen has populated the C# HeroStore.  Load the authoritative /24
    -- snapshot so location actions can select an eligible owned hero.
    HeroController.requestSnapshot("YouLi")
    requestInfo()
    Bridge:SetStatus("YouLi prerequisites /24 and /335 op=1 requested from current WanFa route.")
end

function M.start(locationId, heroId, mode, durationType)
    local message = Protocol.newMessage(Protocol.YOU_LI)
    message:WriteByte(2); message:WriteByte(1); message:WriteWord(heroId)
    message:WriteByte(locationId); message:WriteByte(mode); message:WriteByte(durationType)
    Bridge:Send(message); Bridge:SetStatus("YouLi/335 op=2 requested location="..tostring(locationId).." hero="..tostring(heroId)..".")
end

function M.startBatch(payload)
    local entries = {}
    for locationId, heroId, mode, durationType in string.gmatch(payload or "", "(%d+),(%d+),(%d+),(%d+)") do
        table.insert(entries, {
            locationId = tonumber(locationId), heroId = tonumber(heroId),
            mode = tonumber(mode), durationType = tonumber(durationType)
        })
    end
    if #entries == 0 then Bridge:SetYouLiError("没有可批量派遣的神将"); return end
    local message = Protocol.newMessage(Protocol.YOU_LI)
    message:WriteByte(2); message:WriteByte(#entries)
    for _, entry in ipairs(entries) do
        message:WriteWord(entry.heroId); message:WriteByte(entry.locationId)
        message:WriteByte(entry.mode); message:WriteByte(entry.durationType)
    end
    Bridge:Send(message)
    Bridge:SetStatus("YouLi/335 op=2 batch requested count="..tostring(#entries)..".")
end

function M.claim(locationId)
    local message = Protocol.newMessage(Protocol.YOU_LI)
    message:WriteByte(3); message:WriteByte(1); message:WriteByte(locationId)
    Bridge:Send(message); Bridge:SetStatus("YouLi/335 op=3 requested location="..tostring(locationId)..".")
end

function M.onPacket(command, message)
    if command ~= Protocol.YOU_LI then return false end
    local op = message:ReadByte()
    if op == 1 then
        local count = message:ReadByte()
        Bridge:BeginYouLiUpdate(count)
        for _ = 1, count do
            local id = message:ReadByte(); local mode = message:ReadByte(); local durationType = message:ReadByte()
            local heroId = message:ReadWord(); local lastTime = message:ReadUInt(); local endTime = message:ReadUInt(); local fragments = message:ReadWord()
            local rewardBatches = message:ReadByte()
            for _ = 1, rewardBatches do skipRewardBatch(message) end
            local dialogues = message:ReadByte()
            for _ = 1, dialogues do message:ReadWord() end
            Bridge:AddYouLiRecord(id, mode, durationType, heroId, lastTime, endTime, fragments, rewardBatches, dialogues)
        end
        Bridge:EndYouLiUpdate()
        Bridge:SetStatus("YouLi/335 op=1 recv: records=" .. tostring(count) .. ".")
        if M.validation then M.validation = false; Bridge:CompleteYouLiValidation() end
    elseif op == 2 or op == 3 then
        local success = message:ReadByte()
        if success == 0 and message.Remaining > 0 then Bridge:SetYouLiError(message:ReadString())
        else Bridge:SetStatus("YouLi/335 op="..tostring(op).." succeeded; refreshing authoritative state."); requestInfo() end
    end
    return true
end

return M
