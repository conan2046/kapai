function OnPacket(command, message)
    if command == Protocol.PET_EQUIP_OPERATE then
        local handled, operation = PetEquipRouter:dispatch(message)
        if handled then return end
        Bridge:SetStatus("Unhandled /319 op=" .. tostring(operation) .. ".")
        return
    end
    if FundsController.onPacket(command, message) then return end
    if command == Protocol.TEMP_ACTIVITY then TempActivityController.onPacket(message); return end
    if LoginController.onPacket(command, message) then return end
    if StaminaClaimController.onPacket(command, message) then return end
    if ResourceRecoveryController.onPacket(command, message) then return end
    if PlayerController.onPacket(command, message) then return end
    if BagController.onPacket(command, message) then return end
    if HeroController.onPacket(command, message) then return end
    if SevenDayController.onPacket(command, message) then return end
    if ServerTimeController.onPacket(command, message) then return end
    if MailController.onPacket(command, message) then return end
    if GameplayShopController.onPacket(command, message) then return end
    if ShopController.onPacket(command, message) then return end
    if FriendController.onPacket(command, message) then return end
    if ChatController.onPacket(command, message) then return end
    if TeamController.onPacket(command, message) then return end
    if GuildController.onPacket(command, message) then return end
    if FengShenStoryController.onPacket(command, message) then return end
    if ArenaController.onPacket(command, message) then return end
    if KunLunController.onPacket(command, message) then return end
    if BloodFightController.onPacket(command, message) then return end
    if WorldController.onPacket(command, message) then return end
    if WelfareController.onPacket(command, message) then return end
    if DrawController.onPacket(command, message) then return end
    if HotPointController.onPacket(command, message) then return end
    if YouLiController.onPacket(command, message) then return end
    TaskController.onPacket(command, message)
end
