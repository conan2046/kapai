local Protocol = require("Protocol")
local EquipmentModel = require("Hero.LegacyEquipmentModel")
local BagController = require("Bag.BagController")

local M = {
    requested = false,
    equipmentComplete = false,
    faBaoComplete = false,
    stage = nil,
    injectedEquipment = false,
    injectedFaBao = false,
    equipmentUid = 0,
    faBaoUid = 0,
    formationPosition = 1,
    strengthBefore = 0,
    pendingEquipment = {},
    pendingFaBao = {},
    invalidReason = "",
    repeatReason = "",
    openKind = 1,
    openPending = false,
    preloadCallback = nil,
    attributePushCount = 0,
    attributePushChanged = false,
    mutationHeroId = 0,
    heroPowerBefore = 0,
    heroAttackBefore = 0,
    heroHealthBefore = 0,
    playerPowerBefore = 0,
    g4Started = false,
}

local function showBusinessError(reason)
    local text = tostring(reason or "操作失败")
    text = string.gsub(text, "%[/?c%]", "")
    Bridge:SetStatus(text)
    Bridge:ShowToast(text, 2.0)
end

local function sendOp(op)
    local message = Protocol.newMessage(Protocol.PET_EQUIP_OPERATE)
    message:WriteByte(op)
    Bridge:Send(message)
end

local function requestLists()
    M.equipmentComplete = false
    M.faBaoComplete = false
    sendOp(1)
    sendOp(17)
    Bridge:SetStatus("HeroEquipment/319 list requests sent: op=1, op=17.")
end

local function readEquipmentRecord(message)
    local uid = message:ReadUInt()
    if uid == 0 then return end
    local templateId = message:ReadWord()
    local formationPosition = message:ReadByte()
    local experience = message:ReadUInt()
    local cultivationCount = message:ReadByte()
    local cultivation = {}
    for index = 1, cultivationCount do
        cultivation[index] = { message:ReadByte(), message:ReadWord() }
    end
    local baseAttributeType = message:ReadWord()
    local baseAttributeValue = message:ReadUInt()
    local strengthAttributeType, strengthAttributeValue = 0, 0
    for group = 1, 4 do
        message:ReadByte()
        local attributeCount = message:ReadByte()
        for attribute = 1, attributeCount do
            local attrType, attrValue = message:ReadWord(), message:ReadUInt()
            if group == 1 and attribute == 1 then
                strengthAttributeType, strengthAttributeValue = attrType, attrValue
            end
        end
    end
    return {
        uid = uid, templateId = templateId, fpos = formationPosition, experience = experience,
        cultivation = cultivation, baseAttributeType = baseAttributeType,
        baseAttributeValue = baseAttributeValue, part = Bridge:GetEquipmentPart(templateId),
        strengthAttributeType = strengthAttributeType, strengthAttributeValue = strengthAttributeValue,
    }
end

local function readFaBaoRecord(message)
    local uid = message:ReadUInt()
    if uid == 0 then return end
    local templateId = message:ReadWord()
    local formationPosition = message:ReadByte()
    local slot = message:ReadByte()
    local experience = message:ReadUInt()
    local cultivationCount = message:ReadByte()
    local cultivation = {}
    for index = 1, cultivationCount do
        cultivation[index] = { message:ReadByte(), message:ReadByte() }
    end
    return {
        uid = uid, templateId = templateId, fpos = formationPosition, wpos = slot,
        experience = experience, cultivation = cultivation,
    }
end

-- Publish a complete render snapshot only after the Lua authority has changed.
local function syncRenderMirror()
    Bridge:BeginHeroEquipmentUpdate(EquipmentModel:GetEquipmentCount())
    for _, uid in ipairs(EquipmentModel.equipmentOrder) do
        local value = EquipmentModel.equipment[uid]
        Bridge:BeginHeroEquipmentRecord(value.uid, value.templateId, value.fpos, value.experience,
            value.baseAttributeType, value.baseAttributeValue, value.strengthAttributeType, value.strengthAttributeValue)
        for _, level in ipairs(value.cultivation) do Bridge:AddHeroEquipmentCultivation(level[1], level[2]) end
        Bridge:EndHeroEquipmentRecord()
    end
    Bridge:EndHeroEquipmentUpdate()

    Bridge:BeginFaBaoUpdate(EquipmentModel:GetFaBaoCount())
    for _, uid in ipairs(EquipmentModel.faBaoOrder) do
        local value = EquipmentModel.faBao[uid]
        Bridge:BeginFaBaoRecord(value.uid, value.templateId, value.fpos, value.wpos, value.experience)
        for _, level in ipairs(value.cultivation) do Bridge:AddFaBaoCultivation(level[1], level[2]) end
        Bridge:EndFaBaoRecord()
    end
    Bridge:EndFaBaoUpdate()
end

local function selectFormationPosition()
    local count = Bridge:GetFormationCombatCount()
    for position = 1, count do
        if Bridge:GetFormationHeroAt(position) > 0 then return position end
    end
    return 0
end

local function inject(kind, templateId)
    local message = Protocol.newMessage(Protocol.INTERACT)
    message:WriteByte(53)
    message:WriteByte(kind)
    message:WriteWord(templateId)
    Bridge:Send(message)
end

local function sendEquipmentWear(uid, position)
    local message = Protocol.newMessage(Protocol.PET_EQUIP_OPERATE)
    message:WriteByte(2)
    message:WriteByte(position)
    message:WriteUInt(uid)
    Bridge:Send(message)
end

local function sendEquipmentTakeOff(uid, position)
    local message = Protocol.newMessage(Protocol.PET_EQUIP_OPERATE)
    message:WriteByte(3)
    message:WriteByte(position)
    message:WriteUInt(uid)
    Bridge:Send(message)
end

local function sendEquipmentStrength(uid, strengthType)
    local message = Protocol.newMessage(Protocol.PET_EQUIP_OPERATE)
    message:WriteByte(4)
    message:WriteUInt(uid)
    message:WriteByte(strengthType == 1 and 1 or 0)
    Bridge:Send(message)
end

local function sendFaBaoWear(uid, position, slot)
    local message = Protocol.newMessage(Protocol.PET_EQUIP_OPERATE)
    message:WriteByte(18)
    message:WriteUInt(uid)
    message:WriteByte(position)
    message:WriteByte(slot)
    Bridge:Send(message)
end

local function sendFaBaoTakeOff(uid)
    local message = Protocol.newMessage(Protocol.PET_EQUIP_OPERATE)
    message:WriteByte(19)
    message:WriteUInt(uid)
    Bridge:Send(message)
end

local function sendEquipmentStrengthAll(position)
    local message = Protocol.newMessage(Protocol.PET_EQUIP_OPERATE)
    message:WriteByte(12)
    message:WriteByte(position)
    Bridge:Send(message)
end

local function sendEquipmentRefine(uid, itemId, itemCount)
    local message = Protocol.newMessage(Protocol.PET_EQUIP_OPERATE)
    message:WriteByte(13)
    message:WriteUInt(uid)
    message:WriteByte(1)
    message:WriteWord(itemId)
    message:WriteWord(itemCount)
    Bridge:Send(message)
end

local function sendEquipmentAutoRefine(uid, ...)
    local values = {...}
    local pairs = {}
    for index = 1, 8, 2 do
        local itemId = tonumber(values[index]) or 0
        local itemCount = tonumber(values[index + 1]) or 0
        if itemId > 0 and itemCount > 0 then
            table.insert(pairs, {itemId, itemCount})
        end
    end
    if #pairs == 0 then return end
    local message = Protocol.newMessage(Protocol.PET_EQUIP_OPERATE)
    message:WriteByte(13)
    message:WriteUInt(uid)
    message:WriteByte(#pairs)
    for _, pair in ipairs(pairs) do
        message:WriteWord(pair[1])
        message:WriteWord(pair[2])
    end
    Bridge:Send(message)
end

local function sendEquipmentCultivate(uid, op)
    local message = Protocol.newMessage(Protocol.PET_EQUIP_OPERATE)
    message:WriteByte(op)
    message:WriteUInt(uid)
    message:WriteWord(1)
    Bridge:Send(message)
end

local function sendEquipmentCompose(fragmentId)
    local message = Protocol.newMessage(Protocol.PET_EQUIP_OPERATE)
    message:WriteByte(11)
    message:WriteWord(fragmentId)
    Bridge:Send(message)
end

local function completeMutationOrBeginInvalid()
    if Bridge:HasCommandLineFlag("-projectXHeroEquipG4Validation") then
        M.stage = "await_invalid_equipment"
        sendEquipmentWear(4294967295, M.formationPosition)
    else
        M.stage = nil
        Bridge:CompleteHeroEquipmentMutationValidation(M.equipmentUid, M.strengthBefore,
            Bridge:GetHeroEquipmentStrengthLevel(M.equipmentUid), M.faBaoUid, M.formationPosition, "", "",
            M.attributePushCount, M.attributePushChanged, M.mutationHeroId, M.heroPowerBefore,
            M.heroAttackBefore, M.heroHealthBefore, M.playerPowerBefore)
    end
end

local function beginFaBaoMutation()
    M.stage = "await_fabao_wear"
    sendFaBaoWear(M.faBaoUid, M.formationPosition, 5)
end

local function startMutation()
    local equipment = EquipmentModel:GetFirstEquipment(true)
    local faBao = EquipmentModel:GetFirstFaBao(true)
    M.equipmentUid = equipment and equipment.uid or 0
    M.faBaoUid = faBao and faBao.uid or 0
    M.formationPosition = selectFormationPosition()
    M.mutationHeroId = Bridge:GetFormationHeroAt(M.formationPosition)
    if M.equipmentUid == 0 or M.faBaoUid == 0 or M.formationPosition == 0 then
        Bridge:Fail("Hero equipment mutation requires one unworn equipment, one unworn fabao and an occupied formation position.")
        return
    end
    local originalEquipment = EquipmentModel:GetEquipmentAt(M.formationPosition, equipment.part)
    local originalFaBao = EquipmentModel:GetFaBaoAt(M.formationPosition, 5)
    M.originalEquipmentUid = originalEquipment and originalEquipment.uid or 0
    M.originalFaBaoUid = originalFaBao and originalFaBao.uid or 0
    local strength = 0
    for _, value in ipairs(equipment.cultivation) do if value[1] == 1 then strength = value[2] end end
    M.strengthBefore = strength
    M.attributePushCount = 0
    M.attributePushChanged = false
    M.heroPowerBefore = Bridge:GetHeroPower(M.mutationHeroId)
    M.heroAttackBefore = Bridge:GetHeroAttack(M.mutationHeroId)
    M.heroHealthBefore = Bridge:GetHeroHealth(M.mutationHeroId)
    M.playerPowerBefore = Bridge:GetPlayerPower()
    if Bridge:HasCommandLineFlag("-projectXHeroEquipMaterialValidation") then
        Bridge:SetCurrency(60000, 0)
        local cost = Bridge:GetEquipmentStrengthCost(M.equipmentUid)
        if Bridge:GetCurrency(60000) < cost then
            Bridge:CompleteHeroEquipmentMaterialValidation(M.equipmentUid, M.strengthBefore, "您的金币不足！")
            return
        end
    end
    M.stage = "await_equipment_wear"
    sendEquipmentWear(M.equipmentUid, M.formationPosition)
end

local function finishLists()
    if not M.equipmentComplete or not M.faBaoComplete then return end
    if M.preloadCallback ~= nil then
        local callback = M.preloadCallback
        M.preloadCallback = nil
        callback()
        return
    end
    if Bridge:HasCommandLineFlag("-projectXHeroEquipG4Validation") then
        if not M.g4Started then
            M.g4Started = true
            Bridge:RunHeroEquipmentG4Validation()
        elseif M.openPending then
            M.openPending = false
            Bridge:ShowHeroEquipment(M.openKind)
        end
        return
    end
    if M.stage == "await_final_lists" then
        M.stage = nil
        Bridge:CompleteHeroEquipmentMutationValidation(M.equipmentUid, M.strengthBefore,
            Bridge:GetHeroEquipmentStrengthLevel(M.equipmentUid), M.faBaoUid, M.formationPosition,
            M.invalidReason, M.repeatReason, M.attributePushCount, M.attributePushChanged,
            M.mutationHeroId, M.heroPowerBefore, M.heroAttackBefore, M.heroHealthBefore,
            M.playerPowerBefore)
        return
    end
    if not Bridge:HasCommandLineFlag("-projectXHeroEquipMutationValidation")
        and not Bridge:HasCommandLineFlag("-projectXHeroEquipMaterialValidation") then
        if Bridge:HasCommandLineFlag("-projectXHeroEquipValidation") then Bridge:CompleteHeroEquipmentReadValidation()
        elseif M.openPending then M.openPending = false; Bridge:ShowHeroEquipment(M.openKind) end
        return
    end
    if Bridge.HeroEquipmentCount == 0 or Bridge.FaBaoCount == 0 then
        M.stage = "await_injection"
        if Bridge.HeroEquipmentCount == 0 then inject(1, 1001) else M.injectedEquipment = true end
        if Bridge.FaBaoCount == 0 then inject(2, 1001) else M.injectedFaBao = true end
        return
    end
    startMutation()
end

local function readList(message, kind)
    local total = message:ReadWord()
    if kind == 1 then
        if total == 0 then EquipmentModel:ReplaceEquipment({}); syncRenderMirror(); M.equipmentComplete = true; finishLists(); return end
    else
        if total == 0 then EquipmentModel:ReplaceFaBao({}); syncRenderMirror(); M.faBaoComplete = true; finishLists(); return end
    end
    local packetCount = message:ReadByte()
    local packetIndex = message:ReadByte()
    local itemCount = message:ReadByte()
    if packetIndex == 0 then
        if kind == 1 then M.pendingEquipment = {} else M.pendingFaBao = {} end
    end
    for _ = 1, itemCount do
        if kind == 1 then M.pendingEquipment[#M.pendingEquipment + 1] = readEquipmentRecord(message)
        else M.pendingFaBao[#M.pendingFaBao + 1] = readFaBaoRecord(message) end
    end
    if message.Remaining ~= 0 then Bridge:Fail("HeroEquipment/319 list has unread bytes: " .. tostring(message.Remaining)); return end
    if packetIndex == packetCount - 1 then
        if kind == 1 then EquipmentModel:ReplaceEquipment(M.pendingEquipment); M.equipmentComplete = true
        else EquipmentModel:ReplaceFaBao(M.pendingFaBao); M.faBaoComplete = true end
        syncRenderMirror()
        finishLists()
    end
end

function M.onHeroReady()
    M.requested = true
    M.stage = nil
    M.injectedEquipment = false
    M.injectedFaBao = false
    requestLists()
end

function M.preload(callback)
    M.preloadCallback = callback
    BagController.requestEquipmentSnapshot(function() requestLists() end)
end

function M.wearEquipment(uid, position) sendEquipmentWear(uid, position) end
function M.takeOffEquipment(uid, position) sendEquipmentTakeOff(uid, position) end
function M.strengthEquipment(uid, strengthType)
    if not EquipmentModel:GetEquipment(uid) then return end
    local cost = Bridge:GetEquipmentStrengthCost(uid)
    if Bridge:GetCurrency(60000) < cost then
        Bridge:SetStatus("您的金币不足！")
        Bridge:ShowToast("您的金币不足！", 2.0)
        return
    end
    sendEquipmentStrength(uid, strengthType)
end
function M.strengthAllEquipment(position) sendEquipmentStrengthAll(position) end
function M.refineEquipment(uid, itemId, itemCount)
    if not EquipmentModel:GetEquipment(uid) or itemId <= 0 or itemCount <= 0 then return end
    sendEquipmentRefine(uid, itemId, itemCount)
end
function M.autoRefineEquipment(uid, ...)
    if not EquipmentModel:GetEquipment(uid) then return end
    sendEquipmentAutoRefine(uid, ...)
end
function M.awakenEquipment(uid)
    if EquipmentModel:GetEquipment(uid) then sendEquipmentCultivate(uid, 14) end
end
function M.divineEquipment(uid)
    if EquipmentModel:GetEquipment(uid) then sendEquipmentCultivate(uid, 15) end
end
function M.composeEquipment(fragmentId)
    if fragmentId > 0 then sendEquipmentCompose(fragmentId) end
end

function M.open(kind)
    M.openKind = kind == 2 and 2 or 1
    M.openPending = true
    BagController.requestEquipmentSnapshot(function() requestLists() end)
end
function M.wearFaBao(uid, position) sendFaBaoWear(uid, position, EquipmentModel:SelectFaBaoSlot(position)) end
function M.takeOffFaBao(uid) sendFaBaoTakeOff(uid) end

local function readEquipmentWearResponse(message, op)
    local position = message:ReadByte()
    local uid = message:ReadUInt()
    if message.Remaining < 1 then Bridge:Fail("Equipment wear/takeoff returned no result byte (system gate or incomplete response).") M.stage = nil return end
    local success = message:ReadByte()
    if success == 0 then
        local reason = message.Remaining > 0 and message:ReadString() or "unknown rejection"
        if M.stage == "await_invalid_equipment" and op == 2 then
            M.invalidReason = reason
            M.stage = "await_repeat_fabao_takeoff"
            sendFaBaoTakeOff(M.faBaoUid)
            return
        end
        showBusinessError(reason); M.stage = nil; return
    end
    if message.Remaining > 0 then message:ReadString() end
    EquipmentModel:SetEquipmentFormation(uid, op == 2 and position or 0)
    syncRenderMirror()
    if M.stage == "await_equipment_wear" and op == 2 then
        M.stage = "await_equipment_strength"
        sendEquipmentStrength(uid)
    elseif M.stage == "await_equipment_takeoff" and op == 3 then
        if M.originalEquipmentUid > 0 then
            M.stage = "await_equipment_restore"
            sendEquipmentWear(M.originalEquipmentUid, M.formationPosition)
        else
            beginFaBaoMutation()
        end
    elseif M.stage == "await_equipment_restore" and op == 2 then
        beginFaBaoMutation()
    end
end

local function readEquipmentStrengthResponse(message)
    local uid = message:ReadUInt()
    message:ReadByte()
    if message.Remaining < 1 then Bridge:Fail("Equipment strength returned no result byte (system gate or incomplete response).") M.stage = nil return end
    local success = message:ReadByte()
    if success == 0 then
        local reason = message.Remaining > 0 and message:ReadString() or "unknown rejection"
        if Bridge:HasCommandLineFlag("-projectXHeroEquipMaterialValidation") then
            M.stage = nil
            Bridge:CompleteHeroEquipmentMaterialValidation(uid, M.strengthBefore, reason)
            return
        end
        showBusinessError(reason); M.stage = nil; return
    end
    message:ReadByte()
    message:ReadWord()
    local attributeCount = message:ReadByte()
    for _ = 1, attributeCount do message:ReadWord(); message:ReadUInt() end
    local equipment = EquipmentModel:GetEquipment(uid)
    local strengthAfter = 0
    for _, value in ipairs(equipment and equipment.cultivation or {}) do if value[1] == 1 then strengthAfter = value[2] end end
    if strengthAfter <= M.strengthBefore then
        Bridge:Fail("Equipment op=16 update did not advance strength before op=4 response.")
        M.stage = nil
        return
    end
    if M.stage == "await_equipment_strength" then
        M.stage = "await_equipment_takeoff"
        sendEquipmentTakeOff(uid, M.formationPosition)
    end
end

local function readSimpleEquipmentCultivationResponse(message, op)
    local uid = message:ReadUInt()
    if op == 14 or op == 15 then message:ReadWord() end
    if message.Remaining < 1 then
        Bridge:Fail("Equipment cultivation op=" .. tostring(op) .. " returned no result byte.")
        return
    end
    local success = message:ReadByte()
    if success == 0 then
        showBusinessError(message.Remaining > 0 and message:ReadString() or "unknown rejection")
        return
    end
    if message.Remaining >= 2 then message:ReadWord() end
    if message.Remaining >= 1 then
        local attributeCount = message:ReadByte()
        for _ = 1, attributeCount do
            if message.Remaining >= 6 then message:ReadWord(); message:ReadUInt() end
        end
    end
    if op == 15 and message.Remaining >= 4 then message:ReadWord(); message:ReadWord() end
    Bridge:SetStatus(op == 13 and "装备精炼成功" or op == 14 and "装备觉醒成功" or "装备神铸成功")
    Bridge:ShowToast(op == 13 and "装备精炼成功" or op == 14 and "装备觉醒成功" or "装备神铸成功", 2.0)
    -- The cultivation response and the op=16 equipment push are two
    -- independent packets.  Refresh the authoritative list after a successful
    -- write so a delayed or missing push cannot leave the Lua owner stale.
    M.equipmentComplete = false
    M.faBaoComplete = false
    sendOp(1)
end

local function readEquipmentComposeResponse(message)
    message:ReadWord()
    if message.Remaining < 1 then Bridge:Fail("Equipment compose returned no result byte."); return end
    local success = message:ReadByte()
    if success == 0 then showBusinessError(message.Remaining > 0 and message:ReadString() or "unknown rejection"); return end
    Bridge:SetStatus("装备合成成功")
    Bridge:ShowToast("装备合成成功", 2.0)
end

local function readEquipmentStrengthAllResponse(message)
    message:ReadByte()
    if message.Remaining < 1 then Bridge:Fail("Equipment strengthen-all returned no result byte."); return end
    local success = message:ReadByte()
    if success == 0 then showBusinessError(message.Remaining > 0 and message:ReadString() or "unknown rejection"); return end
    local count = message:ReadByte()
    for _ = 1, count do message:ReadByte(); message:ReadUInt(); message:ReadByte() end
    Bridge:SetStatus("一键强化完成")
end

local function readEquipmentMasterResponse(message, op)
    local position = message:ReadByte()
    if op == 24 then
        local masterType, level = message:ReadByte(), message:ReadByte()
        Bridge:SetStatus("装备大师更新：阵位" .. tostring(position) .. " 类型" .. tostring(masterType) .. " 等级" .. tostring(level))
        return
    end
    local count = message:ReadByte()
    for _ = 1, count do message:ReadByte(); message:ReadByte() end
end

local function readFaBaoWearResponse(message)
    local uid = message:ReadUInt()
    local position = message:ReadByte()
    local slot = message:ReadByte()
    if message.Remaining < 1 then Bridge:Fail("FaBao wear returned no result byte (system gate or incomplete response).") M.stage = nil return end
    local success = message:ReadByte()
    if success == 0 then
        local reason = message.Remaining > 0 and message:ReadString() or "unknown rejection"
        showBusinessError(reason)
        M.stage = nil
        return
    end
    local replacedUid = message:ReadUInt()
    if message.Remaining > 0 then message:ReadString() end
    EquipmentModel:SetFaBaoFormation(uid, position, slot, replacedUid)
    syncRenderMirror()
    if M.stage == "await_fabao_wear" then
        M.stage = "await_fabao_takeoff"
        sendFaBaoTakeOff(uid)
    elseif M.stage == "await_fabao_restore" then
        completeMutationOrBeginInvalid()
    end
end

local function readFaBaoTakeOffResponse(message)
    local uid = message:ReadUInt()
    local success = message:ReadByte()
    if success == 0 then
        local reason = message.Remaining > 0 and message:ReadString() or "unknown rejection"
        if M.stage == "await_repeat_fabao_takeoff" then
            M.repeatReason = reason
            M.stage = "await_final_lists"
            requestLists()
            return
        end
        showBusinessError(reason); M.stage = nil; return
    end
    if message.Remaining > 0 then message:ReadString() end
    EquipmentModel:SetFaBaoFormation(uid, 0, 0, 0)
    syncRenderMirror()
    if M.stage == "await_fabao_takeoff" then
        if M.originalFaBaoUid > 0 then
            M.stage = "await_fabao_restore"
            sendFaBaoWear(M.originalFaBaoUid, M.formationPosition, 5)
        else
            completeMutationOrBeginInvalid()
        end
    end
end

function M.onHeroAttributePush(heroId, power, attack, health, currentHealth)
    if M.stage == nil or heroId ~= M.mutationHeroId then return end
    M.attributePushCount = M.attributePushCount + 1
    if power ~= M.heroPowerBefore or attack ~= M.heroAttackBefore or health ~= M.heroHealthBefore then
        M.attributePushChanged = true
    end
end

function M.onOperation(op, message)
    if op == 1 then readList(message, 1); return true end
    if op == 17 then readList(message, 2); return true end
    if op == 6 then
        EquipmentModel:UpsertEquipment(readEquipmentRecord(message)); syncRenderMirror()
        if M.stage == "await_injection" then M.injectedEquipment = true end
    elseif op == 22 then
        EquipmentModel:UpsertFaBao(readFaBaoRecord(message)); syncRenderMirror()
        if M.stage == "await_injection" then M.injectedFaBao = true end
    elseif op == 16 then EquipmentModel:UpsertEquipment(readEquipmentRecord(message)); syncRenderMirror()
    elseif op == 2 or op == 3 then readEquipmentWearResponse(message, op)
    elseif op == 4 then readEquipmentStrengthResponse(message)
    elseif op == 11 then readEquipmentComposeResponse(message)
    elseif op == 12 then readEquipmentStrengthAllResponse(message)
    elseif op == 13 or op == 14 or op == 15 then readSimpleEquipmentCultivationResponse(message, op)
    elseif op == 24 or op == 25 or op == 26 then readEquipmentMasterResponse(message, op)
    elseif op == 18 then readFaBaoWearResponse(message)
    elseif op == 19 then readFaBaoTakeOffResponse(message)
    else return false end
    if M.stage == "await_injection" and M.injectedEquipment and M.injectedFaBao then
        M.stage = "await_injected_lists"
        requestLists()
    end
    return true
end

function M.onPacket(command, message)
    if command ~= Protocol.PET_EQUIP_OPERATE then return false end
    return M.onOperation(message:ReadByte(), message)
end

function M.reset()
    M.requested, M.equipmentComplete, M.faBaoComplete = false, false, false
    M.stage, M.injectedEquipment, M.injectedFaBao = nil, false, false
    M.equipmentUid, M.faBaoUid, M.strengthBefore = 0, 0, 0
    M.originalEquipmentUid, M.originalFaBaoUid = 0, 0
    M.pendingEquipment, M.pendingFaBao = {}, {}
    M.invalidReason, M.repeatReason = "", ""
    M.openKind = 1
    M.openPending = false
    M.preloadCallback = nil
    M.attributePushCount, M.attributePushChanged, M.mutationHeroId = 0, false, 0
    M.heroPowerBefore, M.heroAttackBefore, M.heroHealthBefore, M.playerPowerBefore = 0, 0, 0, 0
    M.g4Started = false
    EquipmentModel:Reset()
end

return M
