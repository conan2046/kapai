local Protocol = require("Protocol")
local EquipmentController = require("Hero.EquipmentController")
local FormationModel = require("Hero.LegacyFormationModel")
local WorldController = require("World.WorldController")

local M = {
    requested = false,
    mutationStage = nil,
    mutationHero = nil,
    originalPosition = nil,
    targetPosition = nil,
    g4RestoreExpected = false,
    equipmentReady = false,
    formationReady = false,
}

local function requestFormation()
    local message = Protocol.newMessage(Protocol.FORMATION)
    message:WriteByte(1)
    Bridge:Send(message)
end

local function runG4Validation()
    local original, heroId, target = 0, 0, 0
    for position = 1, FormationModel:GetCombatCount() do
        local id = FormationModel:GetCombatHero(position)
        if id > 0 and original == 0 then original = position; heroId = id end
        if id == 0 and target == 0 then target = position end
    end
    if heroId == 0 or target == 0 then
        Bridge:Fail("Hero G4 requires one occupied and one empty formation position.")
        return
    end
    Bridge:RunHeroG4ControlValidation(heroId, original, target)
end

-- Equivalent to the shipped LuaNetSendMsg:QueryFormationInfo chain. The Unity
-- socket/byte stream stays in C#, while request ownership stays in Lua.
function M.onClick()
    M.requested = true
    M.equipmentReady = false
    M.formationReady = false
    if Bridge:HasCommandLineFlag("-projectXHeroG4Validation") then
        EquipmentController.preload(function()
            M.equipmentReady = true
            if M.formationReady then
                M.formationReady = false
                runG4Validation()
            end
        end)
    end
    WorldController.queryHeroEquipmentStages()
    local message = Protocol.newMessage(Protocol.HERO)
    message:WriteByte(1)
    Bridge:Send(message)
    Bridge:SetStatus("Hero/24 list request sent by legacy Lua formation model.")
end

local function sendPosition(heroId, position)
    -- Shipped LuaNetSendMsg:QueryFormationPetPos.
    local message = Protocol.newMessage(Protocol.FORMATION)
    message:WriteByte(4)
    message:WriteWord(heroId)
    message:WriteByte(position)
    Bridge:Send(message)
end

function M.moveHero(heroId, position)
    if M.mutationStage ~= nil then return end
    local original = FormationModel:GetCombatPosition(heroId)
    if original <= 0 or position <= 0 or position == original then return end
    M.mutationHero, M.originalPosition, M.targetPosition = heroId, original, position
    if Bridge:HasCommandLineFlag("-projectXHeroG4Validation") then
        M.mutationStage = M.g4RestoreExpected and "g4_restore_response" or "g4_move_response"
    else
        M.mutationStage = "interactive_response"
    end
    sendPosition(heroId, position)
end

local function readHero(message)
    local hero = {
        id = message:ReadWord(),
        fightPos = message:ReadByte(),
        name = message:ReadString(),
        star = message:ReadByte(),
        breakLevel = message:ReadByte(),
        level = message:ReadWord(),
        experience = message:ReadUInt(),
        maxExperience = message:ReadUInt(),
        power = message:ReadULongInt(),
        attack = message:ReadUInt(),
        physicalDefense = message:ReadUInt(),
        magicDefense = message:ReadUInt(),
        health = message:ReadULongInt(),
        speed = message:ReadUInt(),
    }
    for _ = 1, 23 do message:ReadUInt() end
    message:ReadByte()
    local trainingCount = message:ReadByte()
    for _ = 1, trainingCount do message:ReadByte(); message:ReadWord() end
    local skillCount = message:ReadByte()
    for _ = 1, skillCount do message:ReadByte(); message:ReadWord(); message:ReadWord() end
    return hero
end

local function readHeroList(message)
    local followId = message:ReadWord()
    local count = message:ReadByte()
    local heroes = {}
    for index = 1, count do heroes[index] = readHero(message) end
    if message.Remaining ~= 0 then Bridge:Fail("Hero/24 has unread bytes: " .. tostring(message.Remaining)); return end
    FormationModel:ReplaceHeroes(followId, heroes)
    requestFormation()
end

-- C# stores are now render mirrors only. All authoritative formation state and
-- ordering is calculated by the Lua model before this snapshot is published.
local function syncRenderMirror()
    Bridge:BeginHeroUpdate(FormationModel.followHeroId, FormationModel:GetHeroCount())
    for index = 1, #FormationModel.petlist do
        local hero = FormationModel.petlist[index]
        Bridge:AddHeroRecord(hero.id, hero.fightPos, hero.name, hero.star, hero.breakLevel, hero.level,
            hero.experience, hero.maxExperience, hero.power, hero.attack, hero.physicalDefense,
            hero.magicDefense, hero.health, hero.speed)
    end
    Bridge:EndHeroUpdate()

    Bridge:BeginFormationUpdate(FormationModel.useId, FormationModel:GetFormationCount())
    local formations = FormationModel:GetMyFormations()
    for index = 1, #formations do Bridge:AddFormationRecord(formations[index][1], formations[index][2]) end
    for index = 1, #FormationModel.ShowPosList do
        Bridge:AddFormationDisplayHero(index, FormationModel.ShowPosList[index])
    end
    for index = 1, #FormationModel.CombatPosList do
        Bridge:AddFormationCombatHero(index, FormationModel.CombatPosList[index])
    end
    Bridge:EndFormationUpdate()
end

local function readFormation(message)
    -- Adapted directly from LuaNetRecvdMsg:DealMsgFormation op=1.
    local activeId = message:ReadWord()
    local count = message:ReadByte()
    local formations = {}
    for index = 1, count do formations[index] = { message:ReadWord(), message:ReadByte() } end
    local displayCount = message:ReadByte()
    local display = {}
    for index = 1, displayCount do display[index] = message:ReadWord() end
    local combatCount = message:ReadByte()
    local combat = {}
    for index = 1, combatCount do combat[index] = message:ReadWord() end
    if message.Remaining ~= 0 then Bridge:Fail("Formation/48 has unread bytes: " .. tostring(message.Remaining)); return end

    FormationModel:ReplaceFormation(activeId, formations, display, combat)
    syncRenderMirror()

    if M.mutationStage == "interactive_snapshot" then
        M.mutationStage = nil
        Bridge:SetStatus("Formation position updated by authoritative Lua snapshot.")
        return
    end
    if M.mutationStage == "g4_move_snapshot" then
        if FormationModel:GetCombatHero(M.targetPosition) ~= M.mutationHero then
            Bridge:Fail("G4 real-control move did not persist at the target position.")
            M.mutationStage = nil
            return
        end
        M.mutationStage = nil
        M.g4RestoreExpected = true
        Bridge:ContinueHeroG4FormationRestore(M.mutationHero, M.targetPosition, M.originalPosition)
        return
    elseif M.mutationStage == "g4_restore_snapshot" then
        local restored = FormationModel:GetCombatHero(M.targetPosition)
        if restored ~= M.mutationHero then
            Bridge:Fail("G4 real-control restore did not persist at the original position.")
            M.mutationStage = nil
            return
        end
        M.mutationStage = nil
        M.g4RestoreExpected = false
        Bridge:CompleteHeroG4Validation(M.mutationHero, M.targetPosition, M.originalPosition)
        return
    end

    if Bridge:HasCommandLineFlag("-projectXHeroEquipValidation")
        or Bridge:HasCommandLineFlag("-projectXHeroEquipMutationValidation")
        or Bridge:HasCommandLineFlag("-projectXHeroEquipMaterialValidation") then
        M.requested = false
        EquipmentController.onHeroReady()
        return
    end
    if M.mutationStage == "await_move_snapshot" then
        if FormationModel:GetCombatHero(M.targetPosition) ~= M.mutationHero then
            Bridge:Fail("Lua formation model did not place hero at target position.")
            M.mutationStage = nil
            return
        end
        M.mutationStage = "await_restore_response"
        sendPosition(M.mutationHero, M.originalPosition)
        return
    elseif M.mutationStage == "await_restore_snapshot" then
        local restored = FormationModel:GetCombatHero(M.originalPosition)
        if restored ~= M.mutationHero then
            Bridge:Fail("Lua formation model did not restore hero to original position.")
            M.mutationStage = nil
            return
        end
        M.mutationStage = nil
        Bridge:CompleteFormationLuaMutationValidation(M.mutationHero, M.originalPosition, M.targetPosition, restored)
        return
    end
    if M.requested then
        M.requested = false
        if Bridge:HasCommandLineFlag("-projectXHeroLockedValidation") then
            Bridge:RunHeroLockedControlValidation()
        elseif Bridge:HasCommandLineFlag("-projectXHeroG4Validation") then
            if M.equipmentReady then runG4Validation()
            else M.formationReady = true end
        elseif Bridge:HasCommandLineFlag("-projectXFormationInvalidValidation") then
            M.mutationHero, M.originalPosition, M.targetPosition = 65535, 0, 1
            M.mutationStage = "invalid_response"
            sendPosition(M.mutationHero, M.targetPosition)
        elseif Bridge:HasCommandLineFlag("-projectXFormationMutationValidation") then
            local original, heroId, target = 0, 0, 0
            for position = 1, FormationModel:GetCombatCount() do
                local id = FormationModel:GetCombatHero(position)
                if id > 0 and original == 0 then original = position; heroId = id end
                if id == 0 and target == 0 then target = position end
            end
            if heroId == 0 or target == 0 then Bridge:Fail("Formation mutation requires one occupied and one empty position."); return end
            M.mutationHero, M.originalPosition, M.targetPosition = heroId, original, target
            M.mutationStage = "await_move_response"
            sendPosition(heroId, target)
        else
            Bridge:CompleteHeroLuaReadValidation(FormationModel:GetHeroCount(), FormationModel:GetFormationCount(),
                FormationModel.useId, FormationModel:GetSelectedHeroId())
        end
    end
end

local function readPositionResponse(message)
    local heroId = message:ReadWord()
    local position = message:ReadByte()
    local success = message:ReadByte()
    if success == 0 then
        local reason = message.Remaining > 0 and message:ReadString() or "unknown server rejection"
        if M.mutationStage == "invalid_response" then
            M.mutationStage = nil
            Bridge:CompleteFormationInvalidValidation(heroId, position, reason)
            return
        end
        M.mutationStage = nil
        Bridge:Fail("Formation position change failed: " .. tostring(reason))
        return
    end
    if M.mutationStage == "invalid_response" then
        M.mutationStage = nil
        Bridge:Fail("Invalid formation hero was unexpectedly accepted.")
        return
    end
    if M.mutationStage == "interactive_response" then
        if heroId ~= M.mutationHero or position ~= M.targetPosition then
            M.mutationStage = nil
            Bridge:Fail("Interactive formation response mismatch.")
            return
        end
        M.mutationStage = "interactive_snapshot"
        requestFormation()
        return
    end
    if M.mutationStage == "g4_move_response" and position == M.targetPosition then
        M.mutationStage = "g4_move_snapshot"
    elseif M.mutationStage == "g4_restore_response" and position == M.targetPosition then
        M.mutationStage = "g4_restore_snapshot"
    end
    if heroId ~= M.mutationHero then Bridge:Fail("Formation response hero mismatch."); M.mutationStage = nil; return end
    if M.mutationStage == "await_move_response" and position == M.targetPosition then
        M.mutationStage = "await_move_snapshot"
    elseif M.mutationStage == "await_restore_response" and position == M.originalPosition then
        M.mutationStage = "await_restore_snapshot"
    end
end

function M.onPacket(command, message)
    if command == Protocol.HERO then
        local op = message:ReadByte()
        if op == 1 then readHeroList(message) else return false end
        return true
    end
    if command ~= Protocol.FORMATION then return false end
    local op = message:ReadByte()
    if op == 1 then readFormation(message); return true end
    if op == 4 then readPositionResponse(message); return true end
    return false
end

function M.reset()
    M.requested, M.mutationStage, M.mutationHero = false, nil, nil
    M.originalPosition, M.targetPosition = nil, nil
    M.g4RestoreExpected = false
    M.equipmentReady, M.formationReady = false, false
    FormationModel:Reset()
end

return M
