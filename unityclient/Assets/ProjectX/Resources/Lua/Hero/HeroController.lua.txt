local Protocol = require("Protocol")
local EquipmentController = require("Hero.EquipmentController")
local FormationModel = require("Hero.LegacyFormationModel")
local WorldController = require("World.WorldController")

local M = {
    requested = false,
    mutationStage = nil,
    mutationHero = nil,
    originalPosition = nil,
    targetPosition = nil,
    swapSourceHero = nil,
    swapTargetHero = nil,
    levelUpHero = nil,
    levelUpBeforeLevel = nil,
    cultivationOperation = nil,
    cultivationResponseReady = false,
    cultivationPushReady = false,
    g4RestoreExpected = false,
    equipmentReady = false,
    formationReady = false,
    closureStage = nil,
}

local syncRenderMirror

local function requestFormation()
    local message = Protocol.newMessage(Protocol.FORMATION)
    message:WriteByte(1)
    Bridge:Send(message)
end

local function runG4Validation()
    local original, heroId, target = 0, 0, 0
    for position = 1, FormationModel:GetCombatCount() do
        local id = FormationModel:GetCombatHero(position)
        if id > 0 and original == 0 then original = position; heroId = id end
        if id == 0 and target == 0 then target = position end
    end
    if heroId == 0 or target == 0 then
        Bridge:Fail("Hero G4 requires one occupied and one empty formation position.")
        return
    end
    Bridge:RunHeroG4ControlValidation(heroId, original, target)
end

-- Equivalent to the shipped LuaNetSendMsg:QueryFormationInfo chain. The Unity
-- socket/byte stream stays in C#, while request ownership stays in Lua.
function M.onClick()
    M.requested = true
    M.equipmentReady = false
    M.formationReady = false
    -- The formation screen renders equipment slots too. The legacy bag used
    -- to be the only path requesting /319 op=1/17, so a fresh formation entry
    -- stayed empty until the player opened the bag. Preload the same
    -- authoritative lists for every hero entry and refresh the formation
    -- mirror when both lists arrive.
    EquipmentController.preload(function()
        M.equipmentReady = true
        if M.formationReady then
            M.formationReady = false
            syncRenderMirror()
            if Bridge:HasCommandLineFlag("-projectXHeroG4Validation") then runG4Validation() end
        end
    end)
    WorldController.queryHeroEquipmentStages()
    local message = Protocol.newMessage(Protocol.HERO)
    message:WriteByte(1)
    Bridge:Send(message)
    Bridge:SetStatus("Hero/24 list request sent by legacy Lua formation model.")
end

local function sendPosition(heroId, position)
    -- Shipped LuaNetSendMsg:QueryFormationPetPos.
    local message = Protocol.newMessage(Protocol.FORMATION)
    message:WriteByte(4)
    message:WriteWord(heroId)
    message:WriteByte(position)
    Bridge:Send(message)
end

function M.moveHero(heroId, position)
    if M.mutationStage ~= nil then return end
    local original = FormationModel:GetCombatPosition(heroId)
    if position <= 0 or position == original or FormationModel:GetPetById(heroId) == nil then return end
    M.mutationHero, M.originalPosition, M.targetPosition = heroId, original, position
    if M.closureStage == "mount" then
        M.mutationStage = "closure_move_response"
    elseif Bridge:HasCommandLineFlag("-projectXHeroG4Validation") then
        M.mutationStage = M.g4RestoreExpected and "g4_restore_response" or "g4_move_response"
    else
        M.mutationStage = "interactive_response"
    end
    sendPosition(heroId, position)
end

local function sendPositionSwap(sourcePosition, targetPosition)
    -- Shipped LuaNetSendMsg:QueryFormationChangePos.
    local message = Protocol.newMessage(Protocol.FORMATION)
    message:WriteByte(5)
    message:WriteByte(sourcePosition)
    message:WriteByte(targetPosition)
    Bridge:Send(message)
end

function M.swapFormation(sourcePosition, targetPosition)
    if M.mutationStage ~= nil or sourcePosition == nil or targetPosition == nil then return end
    if sourcePosition <= 0 or targetPosition <= 0 or sourcePosition == targetPosition then return end
    local sourceHero = FormationModel:GetCombatHero(sourcePosition)
    if sourceHero <= 0 then return end
    M.mutationHero = sourceHero
    M.originalPosition, M.targetPosition = sourcePosition, targetPosition
    M.swapSourceHero = sourceHero
    M.swapTargetHero = FormationModel:GetCombatHero(targetPosition)
    if Bridge:HasCommandLineFlag("-projectXHeroG4Validation") then
        M.mutationStage = M.g4RestoreExpected and "g4_restore_response" or "g4_move_response"
    else
        M.mutationStage = "interactive_swap_response"
    end
    sendPositionSwap(sourcePosition, targetPosition)
    Bridge:SetStatus("Formation/48 op=5 swap request sent: "
        .. tostring(sourcePosition) .. "->" .. tostring(targetPosition) .. ".")
end

local function sendFormationAction(op, formationId)
    local message = Protocol.newMessage(Protocol.FORMATION)
    message:WriteByte(op)
    message:WriteWord(formationId)
    Bridge:Send(message)
end

function M.upgradeFormation(formationId)
    if M.mutationStage ~= nil or formationId == nil or formationId <= 0 then return end
    M.mutationStage = "formation_upgrade_response"
    sendFormationAction(2, formationId)
    Bridge:SetStatus("Formation/48 op=2 upgrade request sent: formation=" .. tostring(formationId) .. ".")
end

function M.useFormation(formationId)
    if M.mutationStage ~= nil or formationId == nil or formationId <= 0 then return end
    M.mutationStage = "formation_use_response"
    sendFormationAction(3, formationId)
    Bridge:SetStatus("Formation/48 op=3 use request sent: formation=" .. tostring(formationId) .. ".")
end

-- The closure uses the same replacement-button path as a player. This only
-- arms the existing /48 response assertion; the button callback sends /48.
function M.prepareDrawClosureMount()
    if not Bridge:HasCommandLineFlag("-projectXDrawClosureValidation")
        or M.closureStage ~= "await_material_refresh" then return end
    M.closureStage = "mount"
end

function M.prepareDrawClosureReconnect()
    if Bridge:HasCommandLineFlag("-projectXDrawClosureValidation") and M.closureStage == "mounted" then
        M.closureStage = "await_reconnect"
    end
end

function M.prepareDrawClosureAccountIsolation()
    if Bridge:HasCommandLineFlag("-projectXDrawClosureValidation") and M.closureStage == "await_reconnect" then
        M.closureStage = "await_account_isolation"
    end
end

-- Shipped LuaNetSendMsg:QueryPetLvUp. The screen is refreshed from the
-- authoritative /24 + /48 snapshots; this function never mutates local data.
function M.levelUp(heroId, itemId, itemCount)
    if M.mutationStage ~= nil or heroId == nil or itemId == nil then return end
    if Bridge:HasCommandLineFlag("-projectXDrawClosureValidation")
        and M.closureStage == "await_initial_material" then
        M.closureStage = "levelup"
    end
    local hero = FormationModel:GetPetById(heroId)
    if hero == nil then return end
    M.mutationHero = heroId
    M.levelUpHero = heroId
    M.levelUpBeforeLevel = hero.level
    M.levelUpBeforeExperience = hero.experience
    M.mutationStage = "levelup_response"
    local message = Protocol.newMessage(Protocol.HERO)
    message:WriteByte(3)
    message:WriteWord(heroId)
    message:WriteWord(itemId)
    message:WriteByte(math.max(1, math.min(tonumber(itemCount) or 1, 255)))
    Bridge:Send(message)
    Bridge:SetStatus("Hero/24 op=3 level-up request sent for hero " .. tostring(heroId) .. ".")
end

-- The legacy Cocos controller keeps the selected hero in LCPet. Unity owns the
-- rendered controls, so every interactive selection must flow back into the
-- same Lua model before a later /48 snapshot performs mirror validation.
function M.selectHero(heroId)
    return FormationModel:SelectHero(tonumber(heroId) or 0)
end

local function beginCultivationMutation(name, heroId)
    if M.mutationStage ~= nil or heroId == nil or FormationModel:GetPetById(heroId) == nil then return false end
    M.mutationStage = "cultivation_response"
    M.mutationHero = heroId
    M.cultivationOperation = name
    M.cultivationResponseReady = false
    M.cultivationPushReady = false
    return true
end

local function sendCultivation(op, heroId, extraWriter)
    local message = Protocol.newMessage(Protocol.HERO)
    message:WriteByte(op)
    message:WriteWord(heroId)
    if extraWriter ~= nil then extraWriter(message) end
    Bridge:Send(message)
    Bridge:SetStatus("Hero/24 " .. tostring(M.cultivationOperation) .. " request sent for hero " .. tostring(heroId) .. ".")
end

function M.autoLevelUp(heroId, targetLevel)
    if not beginCultivationMutation("op=4 auto-level", heroId) then return end
    sendCultivation(4, heroId, function(message) message:WriteWord(targetLevel) end)
end

function M.breakUp(heroId)
    if not beginCultivationMutation("op=5 break", heroId) then return end
    sendCultivation(5, heroId)
end

function M.cultivate(heroId, count)
    if not beginCultivationMutation("op=6 cultivate", heroId) then return end
    sendCultivation(6, heroId, function(message)
        message:WriteWord(math.max(1, math.min(tonumber(count) or 1, 65535)))
    end)
end

function M.starUp(heroId)
    if not beginCultivationMutation("op=7 star", heroId) then return end
    sendCultivation(7, heroId)
end

function M.composeHero(fragmentId)
    if M.mutationStage ~= nil or fragmentId == nil or fragmentId <= 0 then return end
    M.mutationStage = "compose_response"
    local message = Protocol.newMessage(Protocol.HERO)
    message:WriteByte(11)
    message:WriteWord(fragmentId)
    Bridge:Send(message)
end

function M.activateCultivation(heroId)
    if not beginCultivationMutation("op=12 activate", heroId) then return end
    sendCultivation(12, heroId)
end

local function readHero(message)
    local hero = {
        id = message:ReadWord(),
        fightPos = message:ReadByte(),
        name = message:ReadString(),
        star = message:ReadByte(),
        breakLevel = message:ReadByte(),
        level = message:ReadWord(),
        experience = message:ReadUInt(),
        maxExperience = message:ReadUInt(),
        power = message:ReadULongInt(),
        attack = message:ReadUInt(),
        physicalDefense = message:ReadUInt(),
        magicDefense = message:ReadUInt(),
        health = message:ReadULongInt(),
        speed = message:ReadUInt(),
        currentHealth = 0,
    }
    for _ = 1, 23 do message:ReadUInt() end
    hero.cultivationLevel = message:ReadByte()
    hero.cultivation = { 0, 0, 0, 0 }
    local trainingCount = message:ReadByte()
    for _ = 1, trainingCount do
        local kind, value = message:ReadByte(), message:ReadWord()
        if kind >= 1 and kind <= 4 then hero.cultivation[kind] = value end
    end
    local skillCount = message:ReadByte()
    hero.primarySkillLevel = 1
    for _ = 1, skillCount do
        local skillPosition = message:ReadByte()
        message:ReadWord()
        local skillLevel = message:ReadWord()
        if skillPosition == 0 then hero.primarySkillLevel = math.max(1, skillLevel) end
    end
    hero.currentHealth = hero.health
    return hero
end

local function readHeroList(message)
    local followId = message:ReadWord()
    local count = message:ReadByte()
    local heroes = {}
    for index = 1, count do heroes[index] = readHero(message) end
    if message.Remaining ~= 0 then Bridge:Fail("Hero/24 has unread bytes: " .. tostring(message.Remaining)); return end
    FormationModel:ReplaceHeroes(followId, heroes)
    requestFormation()
end

local function requestCultivationSnapshotIfReady()
    if M.mutationStage ~= "cultivation_response" or not M.cultivationResponseReady
        or not M.cultivationPushReady then return end
    M.mutationStage = "cultivation_snapshot"
    M.onClick()
end

local function readCultivationResponse(op, message)
    local heroId, success, reason = M.mutationHero or 0, 0, ""
    if op == 4 then
        success = message:ReadByte()
        if success > 0 then heroId = message:ReadWord() else reason = message:ReadString() end
    elseif op == 3 or op == 5 or op == 7 then
        heroId = message:ReadWord(); success = message:ReadByte()
        if success == 0 and message.Remaining > 0 then reason = message:ReadString() end
    elseif op == 6 then
        heroId = message:ReadWord(); message:ReadWord(); success = message:ReadByte()
        if success > 0 then
            local count = message:ReadByte()
            for _ = 1, count do message:ReadByte(); message:ReadWord() end
        elseif message.Remaining > 0 then reason = message:ReadString() end
    elseif op == 12 then
        heroId = message:ReadWord(); success = message:ReadByte()
        if success > 0 then
            message:ReadByte()
            local count = message:ReadByte()
            for _ = 1, count do message:ReadByte(); message:ReadWord() end
        elseif message.Remaining > 0 then reason = message:ReadString() end
    end
    if message.Remaining ~= 0 then
        M.mutationStage = nil
        Bridge:Fail("Hero/24 op=" .. tostring(op) .. " has unread bytes: " .. tostring(message.Remaining))
        return
    end
    if success == 0 then
        M.mutationStage = nil
        Bridge:Fail("Hero cultivation rejected: " .. tostring(reason))
        return
    end
    if heroId ~= M.mutationHero then
        M.mutationStage = nil
        Bridge:Fail("Hero cultivation response hero mismatch: " .. tostring(heroId))
        return
    end
    M.cultivationResponseReady = true
    requestCultivationSnapshotIfReady()
end

local function readFullPetPush(message)
    local op = message:ReadByte()
    if op ~= 2 then return false end
    local hero = readHero(message)
    if message.Remaining ~= 0 then Bridge:Fail("Hero/25 has unread bytes: " .. tostring(message.Remaining)); return true end
    FormationModel:ReplacePet(hero)
    syncRenderMirror()
    if M.mutationStage == "cultivation_response" and hero.id == M.mutationHero then
        M.cultivationPushReady = true
        requestCultivationSnapshotIfReady()
    end
    return true
end

-- C# stores are now render mirrors only. All authoritative formation state and
-- ordering is calculated by the Lua model before this snapshot is published.
syncRenderMirror = function()
    Bridge:BeginHeroUpdate(FormationModel.followHeroId, FormationModel:GetHeroCount())
    for index = 1, #FormationModel.petlist do
        local hero = FormationModel.petlist[index]
        Bridge:AddHeroRecord(hero.id, hero.fightPos, hero.name, hero.star, hero.breakLevel, hero.level,
            hero.experience, hero.maxExperience, hero.power, hero.attack, hero.physicalDefense,
            hero.magicDefense, hero.health, hero.speed, hero.currentHealth or hero.health,
            hero.cultivationLevel or 0,
            hero.cultivation[1] or 0, hero.cultivation[2] or 0,
            hero.cultivation[3] or 0, hero.cultivation[4] or 0,
            hero.primarySkillLevel or 1)
    end
    Bridge:EndHeroUpdate()

    Bridge:BeginFormationUpdate(FormationModel.useId, FormationModel:GetFormationCount())
    local formations = FormationModel:GetMyFormations()
    for index = 1, #formations do Bridge:AddFormationRecord(formations[index][1], formations[index][2]) end
    for index = 1, #FormationModel.ShowPosList do
        Bridge:AddFormationDisplayHero(index, FormationModel.ShowPosList[index])
    end
    for index = 1, #FormationModel.CombatPosList do
        Bridge:AddFormationCombatHero(index, FormationModel.CombatPosList[index])
    end
    Bridge:EndFormationUpdate()
    Bridge:SyncHeroSelection(FormationModel:GetSelectedHeroId())
end

local function readFormation(message)
    -- Adapted directly from LuaNetRecvdMsg:DealMsgFormation op=1.
    local activeId = message:ReadWord()
    local count = message:ReadByte()
    local formations = {}
    for index = 1, count do formations[index] = { message:ReadWord(), message:ReadByte() } end
    local displayCount = message:ReadByte()
    local display = {}
    for index = 1, displayCount do display[index] = message:ReadWord() end
    local combatCount = message:ReadByte()
    local combat = {}
    for index = 1, combatCount do combat[index] = message:ReadWord() end
    if message.Remaining ~= 0 then Bridge:Fail("Formation/48 has unread bytes: " .. tostring(message.Remaining)); return end

    FormationModel:ReplaceFormation(activeId, formations, display, combat)
    syncRenderMirror()
    M.formationReady = true

    if M.mutationStage == "levelup_snapshot" then
        local hero = FormationModel:GetPetById(M.levelUpHero)
        M.requested = false
        if hero == nil or (hero.level <= (M.levelUpBeforeLevel or 0)
            and hero.experience <= (M.levelUpBeforeExperience or 0)) then
            Bridge:Fail("Hero level-up did not persist in the authoritative /24 snapshot.")
            M.mutationStage = nil
            return
        end
        if Bridge:HasCommandLineFlag("-projectXDrawClosureValidation") and M.closureStage == "levelup" then
            M.mutationStage = nil
            M.closureStage = "await_material_refresh"
            Bridge:CompleteDrawClosureHeroLevelUp(hero.id, M.levelUpBeforeLevel or 0, hero.level,
                M.levelUpBeforeExperience or 0, hero.experience)
            Bridge:BeginDrawClosureBagRefresh()
            return
        end
        M.mutationStage = nil
        Bridge:SetStatus("Hero/24 op=3 persisted: hero=" .. tostring(hero.id) .. " level=" .. tostring(hero.level) .. ".")
        return
    end

    if M.mutationStage == "cultivation_snapshot" then
        local operation, heroId = M.cultivationOperation, M.mutationHero
        M.mutationStage, M.mutationHero, M.cultivationOperation = nil, nil, nil
        M.cultivationResponseReady, M.cultivationPushReady = false, false
        Bridge:SetStatus("Hero cultivation authoritative snapshot applied: " .. tostring(operation)
            .. " hero=" .. tostring(heroId) .. ".")
        return
    end

    if M.mutationStage == "closure_move_snapshot" then
        local hero = FormationModel:GetPetById(M.mutationHero)
        M.requested = false
        if hero == nil or FormationModel:GetCombatHero(M.targetPosition) ~= M.mutationHero then
            Bridge:Fail("Draw closure formation snapshot did not contain the recruited hero.")
            M.mutationStage = nil
            return
        end
        M.mutationStage = nil
        M.closureStage = "mounted"
        Bridge:CompleteDrawClosureFormation(hero.id, M.targetPosition, hero.level, hero.attack, hero.health)
        return
    end

    if M.mutationStage == "interactive_swap_snapshot" then
        local sourceAfter = FormationModel:GetCombatHero(M.originalPosition)
        local targetAfter = FormationModel:GetCombatHero(M.targetPosition)
        if targetAfter ~= M.swapSourceHero or sourceAfter ~= (M.swapTargetHero or 0) then
            Bridge:Fail("Formation swap did not persist in the authoritative /48 snapshot.")
            M.mutationStage = nil
            return
        end
        M.mutationStage = nil
        Bridge:SetStatus("Formation positions swapped by authoritative Lua snapshot.")
        return
    end
    if M.mutationStage == "interactive_snapshot" then
        M.mutationStage = nil
        Bridge:SetStatus("Formation position updated by authoritative Lua snapshot.")
        return
    end
    if M.mutationStage == "g4_move_snapshot" then
        if FormationModel:GetCombatHero(M.targetPosition) ~= M.mutationHero then
            Bridge:Fail("G4 real-control move did not persist at the target position.")
            M.mutationStage = nil
            return
        end
        M.mutationStage = nil
        M.g4RestoreExpected = true
        Bridge:ContinueHeroG4FormationRestore(M.mutationHero, M.targetPosition, M.originalPosition)
        return
    elseif M.mutationStage == "g4_restore_snapshot" then
        local restored = FormationModel:GetCombatHero(M.targetPosition)
        if restored ~= M.mutationHero then
            Bridge:Fail("G4 real-control restore did not persist at the original position.")
            M.mutationStage = nil
            return
        end
        M.mutationStage = nil
        M.g4RestoreExpected = false
        Bridge:CompleteHeroG4Validation(M.mutationHero, M.targetPosition, M.originalPosition)
        return
    end

    if Bridge:HasCommandLineFlag("-projectXHeroEquipValidation")
        or Bridge:HasCommandLineFlag("-projectXHeroEquipMutationValidation")
        or Bridge:HasCommandLineFlag("-projectXHeroEquipG4Validation")
        or Bridge:HasCommandLineFlag("-projectXHeroEquipMaterialValidation") then
        M.requested = false
        EquipmentController.onHeroReady()
        return
    end
    if M.mutationStage == "await_move_snapshot" then
        if FormationModel:GetCombatHero(M.targetPosition) ~= M.mutationHero then
            Bridge:Fail("Lua formation model did not place hero at target position.")
            M.mutationStage = nil
            return
        end
        M.mutationStage = "await_restore_response"
        sendPosition(M.mutationHero, M.originalPosition)
        return
    elseif M.mutationStage == "await_restore_snapshot" then
        local restored = FormationModel:GetCombatHero(M.originalPosition)
        if restored ~= M.mutationHero then
            Bridge:Fail("Lua formation model did not restore hero to original position.")
            M.mutationStage = nil
            return
        end
        M.mutationStage = nil
        Bridge:CompleteFormationLuaMutationValidation(M.mutationHero, M.originalPosition, M.targetPosition, restored)
        return
    end
    if M.requested then
        M.requested = false
        if Bridge:HasCommandLineFlag("-projectXDrawClosureValidation") then
            local hero = FormationModel:GetPetById(64)
            if M.closureStage == nil then
                if hero == nil then Bridge:Fail("Draw closure recruited target hero is absent from /24."); return end
                M.closureStage = "await_initial_material"
                Bridge:BeginDrawClosureBagRefresh()
            elseif M.closureStage == "mounted" then
                if hero == nil or FormationModel:GetCombatPosition(hero.id) ~= 1 then
                    Bridge:Fail("Draw closure reentry lost target hero or formation position.")
                    return
                end
                Bridge:CompleteDrawClosureModuleReentry(hero.id, 1, hero.level)
            elseif M.closureStage == "await_reconnect" then
                if hero == nil or FormationModel:GetCombatPosition(hero.id) ~= 1 then
                    Bridge:Fail("Draw closure reconnect lost target hero or formation position.")
                    return
                end
                Bridge:CompleteDrawClosureReconnect(hero.id, 1, hero.level, hero.experience)
            elseif M.closureStage == "await_account_isolation" then
                local target = FormationModel:GetPetById(64)
                local position = target == nil and 0 or FormationModel:GetCombatPosition(64)
                Bridge:CompleteDrawClosureAccountIsolation(FormationModel:GetHeroCount(), position)
            end
        elseif Bridge:HasCommandLineFlag("-projectXHeroLockedValidation") then
            Bridge:RunHeroLockedControlValidation()
        elseif Bridge:HasCommandLineFlag("-projectXHeroCultivationG3Validation") then
            Bridge:RunHeroCultivationG3Validation()
        elseif Bridge:HasCommandLineFlag("-projectXHeroG4Validation") then
            if M.equipmentReady then runG4Validation()
            else M.formationReady = true end
        elseif Bridge:HasCommandLineFlag("-projectXHeroEquipG4Validation")
            or Bridge:HasCommandLineFlag("-projectXEnhanceMasterG3Validation") then
            M.requested = false
            EquipmentController.onHeroReady()
        elseif Bridge:HasCommandLineFlag("-projectXFormationInvalidValidation") then
            M.mutationHero, M.originalPosition, M.targetPosition = 65535, 0, 1
            M.mutationStage = "invalid_response"
            sendPosition(M.mutationHero, M.targetPosition)
        elseif Bridge:HasCommandLineFlag("-projectXFormationMutationValidation") then
            local original, heroId, target = 0, 0, 0
            for position = 1, FormationModel:GetCombatCount() do
                local id = FormationModel:GetCombatHero(position)
                if id > 0 and original == 0 then original = position; heroId = id end
                if id == 0 and target == 0 then target = position end
            end
            if heroId == 0 or target == 0 then Bridge:Fail("Formation mutation requires one occupied and one empty position."); return end
            M.mutationHero, M.originalPosition, M.targetPosition = heroId, original, target
            M.mutationStage = "await_move_response"
            sendPosition(heroId, target)
        else
            Bridge:CompleteHeroLuaReadValidation(FormationModel:GetHeroCount(), FormationModel:GetFormationCount(),
                FormationModel.useId, FormationModel:GetSelectedHeroId())
        end
    end
end

local function readPositionResponse(message)
    local heroId = message:ReadWord()
    local position = message:ReadByte()
    local success = message:ReadByte()
    if success == 0 then
        local reason = message.Remaining > 0 and message:ReadString() or "unknown server rejection"
        if M.mutationStage == "invalid_response" then
            M.mutationStage = nil
            Bridge:CompleteFormationInvalidValidation(heroId, position, reason)
            return
        end
        M.mutationStage = nil
        Bridge:Fail("Formation position change failed: " .. tostring(reason))
        return
    end
    if M.mutationStage == "invalid_response" then
        M.mutationStage = nil
        Bridge:Fail("Invalid formation hero was unexpectedly accepted.")
        return
    end
    if M.mutationStage == nil then
        -- A disconnect can leave an already-acknowledged /48 op=4 packet in the
        -- socket queue. It has no matching in-flight position mutation, so it
        -- must not be treated as a response for the next account/session.
        return
    end
    if M.mutationStage == "interactive_response" then
        if heroId ~= M.mutationHero or position ~= M.targetPosition then
            M.mutationStage = nil
            Bridge:Fail("Interactive formation response mismatch.")
            return
        end
        M.mutationStage = "interactive_snapshot"
        requestFormation()
        return
    end
    if M.mutationStage == "g4_move_response" and position == M.targetPosition then
        M.mutationStage = "g4_move_snapshot"
    elseif M.mutationStage == "g4_restore_response" and position == M.targetPosition then
        M.mutationStage = "g4_restore_snapshot"
    elseif M.mutationStage == "closure_move_response" and position == M.targetPosition then
        M.mutationStage = "closure_move_snapshot"
        requestFormation()
    end
    if heroId ~= M.mutationHero then Bridge:Fail("Formation response hero mismatch."); M.mutationStage = nil; return end
    if M.mutationStage == "await_move_response" and position == M.targetPosition then
        M.mutationStage = "await_move_snapshot"
    elseif M.mutationStage == "await_restore_response" and position == M.originalPosition then
        M.mutationStage = "await_restore_snapshot"
    end
end

local function readPositionSwapResponse(message)
    local sourcePosition = message:ReadByte()
    local targetPosition = message:ReadByte()
    local success = message:ReadByte()
    if success == 0 then
        local reason = message.Remaining > 0 and message:ReadString() or "unknown server rejection"
        M.mutationStage = nil
        Bridge:SetStatus("Formation swap rejected: " .. tostring(reason))
        Bridge:ShowToast(reason, 2)
        return
    end
    if M.mutationStage == nil then return end
    if sourcePosition ~= M.originalPosition or targetPosition ~= M.targetPosition then
        M.mutationStage = nil
        Bridge:Fail("Formation swap response position mismatch.")
        return
    end
    if M.mutationStage == "interactive_swap_response" then
        M.mutationStage = "interactive_swap_snapshot"
    elseif M.mutationStage == "g4_move_response" then
        M.mutationStage = "g4_move_snapshot"
    elseif M.mutationStage == "g4_restore_response" then
        M.mutationStage = "g4_restore_snapshot"
    else
        M.mutationStage = nil
        Bridge:Fail("Unexpected formation swap response stage.")
        return
    end
    requestFormation()
end

local function readFormationUpgradeResponse(message)
    local formationId = message:ReadWord()
    local success = message:ReadByte()
    if success == 0 then
        local reason = message.Remaining > 0 and message:ReadString() or "unknown server rejection"
        M.mutationStage = nil
        Bridge:SetStatus("Formation upgrade rejected: " .. tostring(reason))
        Bridge:ShowToast(reason, 2)
        return
    end
    local level = message:ReadByte()
    FormationModel:AddFormation(formationId, level)
    M.mutationStage = nil
    syncRenderMirror()
    Bridge:SetStatus("Formation/48 op=2 persisted: formation=" .. tostring(formationId)
        .. " level=" .. tostring(level) .. ".")
    Bridge:ShowToast("阵法升级成功", 2)
end

local function readFormationUseResponse(message)

local function readPetUpdate(message)
    local heroId = message:ReadWord()
    local hero = FormationModel:GetPetById(heroId)
    local count = message:ReadWord()
    local values = {}
    for _ = 1, count do
        local kind = message:ReadUInt()
        local value = (kind == 507 or kind == 4) and message:ReadULongInt() or message:ReadUInt()
        if kind == 1 then values.attack = value
        elseif kind == 2 then values.physicalDefense = value
        elseif kind == 3 then values.magicDefense = value
        elseif kind == 4 then values.health = value
        elseif kind == 5 then values.speed = value
        elseif kind == 503 then values.currentHealth = value
        elseif kind == 507 then values.power = value end
    end
    if message.Remaining ~= 0 then
        Bridge:Fail("Hero/70 has unread bytes: " .. tostring(message.Remaining))
        return
    end
    if hero == nil then
        -- The original client silently ignores a push received before /24 has
        -- created this hero. Consuming it avoids leaking a stale packet into a
        -- later account/session while preserving that lifecycle behavior.
        Bridge:SetStatus("Hero/70 ignored before hero snapshot: hero=" .. tostring(heroId) .. ".")
        return
    end
    hero = FormationModel:ApplyPetUpdate(heroId, values)
    syncRenderMirror()
    EquipmentController.onHeroAttributePush(heroId, hero.power, hero.attack, hero.health,
        hero.currentHealth or hero.health)
    Bridge:SetStatus("Hero/70 authoritative attributes applied: hero=" .. tostring(heroId)
        .. " count=" .. tostring(count) .. ".")
end

local function readLevelUpResponse(message)
    local success = message:ReadByte()
    if success == 0 then
        local reason = message.Remaining > 0 and message:ReadString() or "unknown server rejection"
        M.mutationStage = nil
        Bridge:Fail("Hero level-up failed: " .. tostring(reason))
        return
    end
    if M.mutationStage ~= "levelup_response" then return end
    M.mutationStage = "levelup_snapshot"
    M.onClick()
end

local function readHeroComposeResponse(message)
    local fragmentId = message:ReadWord()
    local success = message:ReadByte()
    M.mutationStage = nil
    if success == 0 then
        local reason = message.Remaining > 0 and message:ReadString() or "神将碎片不足"
        Bridge:ShowToast(reason, 2)
        return
    end
    M.onClick()
    Bridge:ShowToast("神将合成成功", 2)
end

function M.onPacket(command, message)
    if command == Protocol.UPDATE_PET_INFO then readPetUpdate(message); return true end
    if command == Protocol.UPDATE_PET then return readFullPetPush(message) end
    if command == Protocol.HERO then
        local op = message:ReadByte()
        if op == 1 then readHeroList(message)
        elseif op == 3 then readLevelUpResponse(message)
        elseif op == 4 or op == 5 or op == 6 or op == 7 or op == 12 then readCultivationResponse(op, message)
        elseif op == 11 then readHeroComposeResponse(message)
        else return false end
        return true
    end
    if command ~= Protocol.FORMATION then return false end
    local op = message:ReadByte()
    if op == 1 then readFormation(message); return true end
    if op == 2 then readFormationUpgradeResponse(message); return true end
    if op == 3 then readFormationUseResponse(message); return true end
    if op == 4 then readPositionResponse(message); return true end
    if op == 5 then readPositionSwapResponse(message); return true end
    return false
end

function M.reset()
    local keepClosure = Bridge:HasCommandLineFlag("-projectXDrawClosureValidation")
        and (M.closureStage == "await_reconnect" or M.closureStage == "await_account_isolation")
    M.requested, M.mutationStage, M.mutationHero = false, nil, nil
    M.originalPosition, M.targetPosition = nil, nil
    M.swapSourceHero, M.swapTargetHero = nil, nil
    M.levelUpHero, M.levelUpBeforeLevel, M.levelUpBeforeExperience = nil, nil, nil
    M.cultivationOperation = nil
    M.cultivationResponseReady, M.cultivationPushReady = false, false
    M.g4RestoreExpected = false
    M.equipmentReady, M.formationReady = false, false
    M.closureStage = keepClosure and M.closureStage or nil
    FormationModel:Reset()
end

return M
