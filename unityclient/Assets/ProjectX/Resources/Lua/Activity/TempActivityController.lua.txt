local Protocol = require("Protocol")
local ActivityController = require("Activity.ActivityController")
local WelfareController = require("Welfare.WelfareController")

local M = {}

function M.requestHudState()
    -- Steam excludes online welfare and the three rotating commercial offers.
    -- Keep packet readers for compatibility with an unsolicited legacy push, but
    -- do not initiate /222 op4 or op89-91 from the Steam HUD.
end

local function readHudDiscount(operation, message)
    local operationType = message:ReadByte()
    if operationType ~= 1 then
        if message.Remaining > 0 then message:ReadByte() end
        if message.Remaining > 0 then message:ReadString() end
        Bridge:SetHudDiscountState(operation, 0, false)
        return true
    end
    local success = message:ReadByte()
    if success == 0 then
        if message.Remaining > 0 then message:ReadString() end
        Bridge:SetHudDiscountState(operation, 0, false)
        return true
    end
    local leftTime = message:ReadUInt()
    message:ReadByte()
    message:ReadUInt()
    message:ReadUInt()
    local bought = message:ReadByte() ~= 0
    Bridge:SetHudDiscountState(operation, leftTime, not bought)
    return true
end

function M.onPacket(message)
    local operation = message:ReadByte()
    if operation >= 89 and operation <= 91 then return readHudDiscount(operation, message) end
    if ActivityController.onTempActivity(operation, message) then return true end
    if WelfareController.onTempActivity(operation, message) then return true end
    Bridge:SetStatus("Unhandled /222 operation: " .. tostring(operation))
    return true
end

return M
