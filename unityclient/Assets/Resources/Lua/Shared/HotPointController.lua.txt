local Protocol = require("Protocol")
local GameplayController = require("Gameplay.GameplayController")
local M = {}

function M.refreshGameplay()
    require("Gameplay.MonopolyController").preload()
    require("Gameplay.MoneyTreeController").preload()
    require("Gameplay.FengShenStoryController").preload()
    require("Gameplay.XunBaoController").preload()
    require("Task.TaskController").preload()
    require("Gameplay.AnswerController").preload()
    require("Gameplay.HappyWheelController").preload()
    require("Gameplay.FishController").preload()
end

function M.onPacket(command, message)
    if command ~= Protocol.FUNC_HOT_POINT then return false end
    local op = message:ReadByte()
    if op ~= 1 and op ~= 2 then
        Bridge:SetStatus("/65 unsupported HUD red-dot op=" .. tostring(op))
        return true
    end
    local kind = message:ReadWord()
    local state = message:ReadByte()
    if message.Remaining ~= 0 then
        Bridge:Fail("/65 has " .. tostring(message.Remaining) .. " unread bytes.")
        return true
    end
    GameplayController.onHotPoint(op, kind, state)
    Bridge:SetHudRedDot(kind, state ~= 0)
    return true
end

return M
