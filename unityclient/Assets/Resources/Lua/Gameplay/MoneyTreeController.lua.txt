local Protocol = require("Protocol")

local M = { pending = 0 }
local MONEY_TREE_OPERATION = 17

local function writeRequest(operation, treeType)
    local message = Protocol.newMessage(Protocol.TEMP_ACTIVITY)
    message:WriteByte(MONEY_TREE_OPERATION)
    message:WriteByte(operation)
    if operation == 2 then message:WriteByte(treeType) end
    Bridge:Send(message)
end

local function readRecord(message, treeType)
    local type = treeType or message:ReadByte()
    return type, message:ReadByte(), message:ReadByte(), message:ReadByte(),
        message:ReadWord(), message:ReadUInt(), message:ReadWord(), message:ReadUInt()
end

function M.onClick(functionId)
    M.pending = 0
    Bridge:ShowMoneyTree(functionId)
    writeRequest(1, 0)
    Bridge:SetStatus("MoneyTree/222 op=17 query requested.")
end

function M.shake(treeType)
    treeType = math.floor(tonumber(treeType) or 0)
    if treeType < 1 or treeType > 2 or M.pending ~= 0 or not Bridge:BeginMoneyTreeShake(treeType) then return false end
    M.pending = treeType
    writeRequest(2, treeType)
    Bridge:SetStatus("MoneyTree/222 op=17 shake requested: type=" .. tostring(treeType) .. ".")
    return true
end

function M.onOperation(operation, message)
    if operation ~= MONEY_TREE_OPERATION then return false end
    local requestType = message:ReadByte()
    if requestType == 1 then
        local success = message:ReadByte()
        if success == 0 then
            Bridge:ClearMoneyTreeState()
            Bridge:ShowToast(message.Remaining > 0 and message:ReadString() or "摇钱树数据同步失败")
            return true
        end
        local count = message:ReadByte()
        Bridge:BeginMoneyTreeState()
        for _ = 1, count do Bridge:AddMoneyTreeRecord(readRecord(message)) end
        if message.Remaining ~= 0 then
            Bridge:ClearMoneyTreeState()
            Bridge:Fail("MoneyTree/222 query has unread bytes=" .. tostring(message.Remaining) .. ")")
            return true
        end
        Bridge:CommitMoneyTreeState()
        Bridge:SetStatus("MoneyTree/222 op=17 query received: records=" .. tostring(count) .. ".")
        return true
    end
    if requestType == 2 then
        local treeType = message:ReadByte()
        local success = message:ReadByte()
        if success == 0 then
            if M.pending ~= 0 then Bridge:FailMoneyTreeShake(M.pending) end
            M.pending = 0
            Bridge:ShowToast(message.Remaining > 0 and message:ReadString() or "摇钱失败")
            return true
        end
        if M.pending == 0 or treeType ~= M.pending then
            if M.pending ~= 0 then Bridge:FailMoneyTreeShake(M.pending) end
            M.pending = 0
            Bridge:Fail("MoneyTree/222 shake identity mismatch: type=" .. tostring(treeType) .. ".")
            return true
        end
        local type, usedCount, freeCount, maxCount, costType, costValue, rewardType, rewardValue = readRecord(message, treeType)
        if message.Remaining ~= 0 then
            Bridge:FailMoneyTreeShake(M.pending)
            M.pending = 0
            Bridge:Fail("MoneyTree/222 shake has unread bytes=" .. tostring(message.Remaining) .. ")")
            return true
        end
        Bridge:CompleteMoneyTreeShake(type, usedCount, freeCount, maxCount, costType, costValue, rewardType, rewardValue)
        M.pending = 0
        Bridge:SetStatus("MoneyTree/222 op=17 shake received: type=" .. tostring(type) .. ", used=" .. tostring(usedCount) .. ".")
        return true
    end
    if M.pending ~= 0 then Bridge:FailMoneyTreeShake(M.pending); M.pending = 0 end
    Bridge:Fail("MoneyTree/222 returned unsupported requestType=" .. tostring(requestType) .. ".")
    return true
end

function M.reset()
    M.pending = 0
    Bridge:ClearMoneyTreeState()
end

return M
