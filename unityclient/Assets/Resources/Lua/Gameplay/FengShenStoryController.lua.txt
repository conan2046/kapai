local Protocol = require("Protocol")
local itemCatalog = require("Data.ItemCatalog")
local itemsById = {}
for _, value in ipairs(itemCatalog) do itemsById[value.id] = value end
local M = { validation = false, validationStarted = false }

local function sendOperation(op)
    local message = Protocol.newMessage(Protocol.WORLD_STAGE)
    message:WriteByte(op)
    Bridge:Send(message)
end

function M.onClick()
    M.validation = Bridge:HasCommandLineFlag("-projectXFengShenStoryValidation") and not M.validationStarted
    M.battleValidation = Bridge:HasCommandLineFlag("-projectXBattleFengShenStoryValidation") and not M.battleValidationStarted
    Bridge:ShowFengShenStory()
    sendOperation(24)
    Bridge:SetStatus("FengShenStory/320 op=24 requested from current WanFa route.")
end

function M.challenge()
    sendOperation(25)
    Bridge:SetStatus("FengShenStory/320 op=25 challenge requested from the current stage popup.")
end

function M.onPacket(command, message)
    if command ~= Protocol.WORLD_STAGE then return false end
    -- Older local servers could emit an op-only /320 response after a
    -- successful op=25 fight. The transport strips the echoed request op,
    -- leaving an empty body. Consume it defensively instead of turning the
    -- entire application state into Failed; current servers send op=25 + 1.
    if message.Remaining == 0 then
        Bridge:SetStatus("FengShenStory/320 ignored legacy empty challenge acknowledgement.")
        return true
    end
    local startPosition = message.Position
    local op = message:ReadByte()
    if op == 24 then
        local chapterId = message:ReadUInt()
        local levelId = message:ReadUInt()
        local count = message:ReadByte()
        Bridge:SetFengShenStoryState(chapterId, levelId, count)
        Bridge:SetStatus("FengShenStory/320 op=24 recv: chapter=" .. tostring(chapterId) .. ", level=" .. tostring(levelId) .. ", count=" .. tostring(count) .. ".")
        if M.battleValidation then
            M.battleValidation = false
            M.battleValidationStarted = true
            Bridge:BeginBattleFengShenStoryValidation()
        elseif M.validation then
            M.validation = false
            M.validationStarted = true
            Bridge:BeginFengShenStoryValidation()
        end
        return true
    elseif op == 25 then
        local succeeded = message:ReadByte() ~= 0
        local errorMessage = ""
        if not succeeded then errorMessage = message:ReadString() end
        Bridge:SetFengShenStoryChallengeResult(succeeded, errorMessage)
        return true
    elseif op == 10 then
        local chapterId = message:ReadUInt()
        local levelId = message:ReadUInt()
        local count = message:ReadByte()
        local unlockedChapterId = message:ReadUInt()
        local unlockedLevelId = message:ReadUInt()
        Bridge:SetFengShenStoryFightPush(chapterId, levelId, count, unlockedChapterId, unlockedLevelId)
        local stars = message:ReadByte()
        local rewardCount = message:ReadByte()
        Bridge:BeginRewardUpdate(rewardCount)
        for _ = 1, rewardCount do
            local rewardType = message:ReadWord()
            local rewardId = message:ReadUInt()
            local amount = message:ReadUInt()
            local displayId = rewardId > 0 and rewardId or rewardType
            local found = itemsById[displayId] or {}
            local name = found.name or ""
            if name == "" then name = "奖励 #" .. tostring(displayId) end
            Bridge:AddRewardRecord(rewardType, rewardId, amount, name, found.pic or 0, found.quality or 0)
        end
        if message.Remaining ~= 0 then
            Bridge:Fail("FengShenStory/320 op=10 unread bytes: " .. tostring(message.Remaining))
            return true
        end
        Bridge:ShowFengShenStoryBattleResult(stars)
        return true
    elseif op == 26 then
        Bridge:ClearFengShenStoryRewardPush()
        local count = message:ReadByte()
        for _ = 1, count do
            local rewardType = message:ReadWord()
            local rewardId = message:ReadUInt()
            local amount = message:ReadUInt()
            local displayId = rewardId > 0 and rewardId or rewardType
            local found = itemsById[displayId] or {}
            local name = found.name or ""
            if name == "" then name = "奖励 #" .. tostring(displayId) end
            Bridge:PushFengShenStoryReward(rewardType, rewardId, amount, name,
                found.pic or 0, found.quality or 0)
        end
        Bridge:ShowFengShenStoryRewardPush()
        return true
    end

    -- /320 is shared with World. Leave unmatched packets untouched.
    message.Position = startPosition
    return false
end

return M
